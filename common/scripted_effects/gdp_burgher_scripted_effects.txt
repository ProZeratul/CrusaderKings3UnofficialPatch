burgher_resolve_county_effect = {
	$FACTION$ = {
		var:burgher_county ?= { save_temporary_scope_as = burgher_county_candidate }

		if = {
			limit = {
				exists = scope:burgher_county_candidate
				any_faction_county_member = { this = scope:burgher_county_candidate }
			}
			scope:burgher_county_candidate = { save_scope_as = burgher_county }
		}
		else = {
			ordered_faction_county_member = {
				order_by = title_province.province_size_level
				save_scope_as = burgher_county
			}
		}

		if = {
			limit = { exists = scope:burgher_county }
			set_variable = {
				name = burgher_county
				value = scope:burgher_county
			}
		}
	}
}

burgher_find_local_notable_effect = {
	$COUNTY$ = { save_temporary_scope_as = notable_county }

	scope:notable_county.holder = {
		random_pool_character = {
			province = scope:notable_county.title_province
			limit = { burgher_can_head_commune_trigger = yes }
			weight = {
				base = 1
				modifier = {
					add = 3
					faith = scope:notable_county.faith
				}
				modifier = {
					add = 3
					culture = scope:notable_county.culture
				}
				modifier = {
					add = 2
					has_trait = education_stewardship
				}
			}
			save_scope_as = burgher_local_notable
		}
	}
}

burgher_find_or_create_local_burgher_effect = {
	$COUNTY$ = { save_temporary_scope_as = local_burgher_county }

	if = {
		limit = {
			scope:local_burgher_county.holder ?= {
				any_pool_character = {
					province = scope:local_burgher_county.title_province
					burgher_can_head_commune_trigger = yes
				}
			}
		}
		burgher_find_local_notable_effect = { COUNTY = scope:local_burgher_county }
		scope:burgher_local_notable ?= { save_scope_as = $SAVE_AS$ }
	}
	else_if = {
		limit = { exists = scope:local_burgher_county.holder }

		create_character = {
			location = scope:local_burgher_county.title_province
			template = burgher_faction_leader_template
			faith = scope:local_burgher_county.faith
			culture = scope:local_burgher_county.culture
			gender_female_chance = scope:local_burgher_county.title_faith_dominant_gender_female_chance
			save_scope_as = $SAVE_AS$
		}

		scope:$SAVE_AS$ = {
			set_character_rite = scope:local_burgher_county.rite
			add_character_flag = $FLAG$
		}
	}
}

setup_burgher_leader_effect = {
	save_scope_as = faction

	burgher_resolve_county_effect = { FACTION = this }

	if = {
		limit = { exists = special_character }
		special_character = { save_scope_as = burgher_spokesman }
	}

	if = {
		limit = {
			NOT = { exists = scope:burgher_spokesman }
			exists = scope:burgher_county
		}

		burgher_find_or_create_local_burgher_effect = {
			COUNTY = scope:burgher_county
			SAVE_AS = burgher_spokesman
			FLAG = burgher_spawned_leader
		}

		if = {
			limit = { exists = scope:burgher_spokesman }
			create_dynamic_title = {
				tier = duchy
				name = FACTION_BURGHER_TITLE_NAME
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = burgher_title_change
				add_claim_on_loss = no
			}
			scope:new_title = {
				set_capital_county = scope:burgher_county
				set_landless_title = yes
				set_destroy_on_succession = yes
				set_delete_on_destroy = yes
				set_no_automatic_claims = yes
				set_definitive_form = yes
				set_can_be_named_after_dynasty = no
				set_can_use_nomadic_naming = no
				change_title_holder = {
					holder = scope:burgher_spokesman
					change = scope:burgher_title_change
				}
				set_variable = {
					name = faction
					value = scope:faction
				}
			}
			resolve_title_and_vassal_change = scope:burgher_title_change
			scope:new_title = { generate_coa = factions }

			scope:burgher_spokesman = {
				set_variable = {
					name = burgher_title
					value = scope:new_title
				}
			}
		}
	}

	if = {
		limit = { exists = scope:burgher_spokesman }

		scope:burgher_spokesman = {
			if = {
				limit = {
					OR = {
						NOT = { exists = joined_faction }
						joined_faction != scope:faction
					}
				}
				join_faction_skip_check = scope:faction
			}
			set_variable = {
				name = burgher_leader_of
				value = scope:faction
			}
		}

		set_special_character = scope:burgher_spokesman

		scope:burgher_spokesman = { save_scope_as = burgher_leader }
	}
}

burgher_resume_pending_leaders_on_war_end_effect = {
	if = {
		limit = { exists = scope:war }
		scope:war = {
			every_war_participant = {
				limit = {
					has_character_flag = burgher_leader_pending_dissolve
					OR = {
						has_character_flag = burgher_spawned_leader
						has_variable = burgher_title
					}
				}
				burgher_resume_pending_leader_dissolve_effect = yes
			}
		}
	}
}

burgher_resume_pending_leader_dissolve_effect = {
	if = {
		limit = {
			has_character_flag = burgher_leader_pending_dissolve
			OR = {
				has_character_flag = burgher_spawned_leader
				has_variable = burgher_title
			}
		}
		remove_character_flag = burgher_leader_pending_dissolve
		burgher_dissolve_leader_effect = yes
	}
}

burgher_dissolve_leader_effect = {
	if = {
		limit = {
			OR = {
				has_character_flag = burgher_spawned_leader
				has_variable = burgher_title
			}
		}

		if = {
			limit = {
				exists = this.var:burgher_title
				trigger_if = {
					limit = { is_at_war = yes }
					any_held_title = { count >= 2 }
				}
			}
			destroy_title = this.var:burgher_title
			remove_variable = burgher_title
		}

		if = {
			limit = {
				has_character_flag = burgher_spawned_leader
				is_alive = yes
				NOT = { has_character_flag = burgher_leader_dissolving }
			}
			remove_character_flag = burgher_leader_pending_dissolve
			add_character_flag = burgher_leader_dissolving
			trigger_event = {
				id = gdp_burgher.0090
				days = 1
			}
		}
		else_if = {
			limit = {
				has_variable = burgher_title
				is_at_war = yes
			}
			add_character_flag = burgher_leader_pending_dissolve
		}
		else = {
			remove_variable = burgher_leader_of
		}
	}
}

burgher_advance_privilege_level_effect = {
	$COUNTY$ = {
		if = {
			limit = { NOT = { has_variable = civic_privileges_level } }
			set_variable = {
				name = civic_privileges_level
				value = 0
			}
		}
		change_variable = {
			name = civic_privileges_level
			add = 1
		}

		if = {
			limit = {
				has_variable = civic_privileges_level
				var:civic_privileges_level >= burgher_max_privilege_level_value
				NOT = { has_variable = civic_league_charter }
			}
			set_variable = civic_league_charter
		}

		if = {
			limit = { has_county_modifier = burgher_grievance_county_modifier }
			remove_county_modifier = burgher_grievance_county_modifier
		}
	}
}

burgher_grant_privilege_tier_effect = {
	if = {
		limit = {
			$COUNTY$ = {
				burgher_lacks_privilege_tier_trigger = { PRIVILEGE = $PRIVILEGE$ }
			}
		}

		$COUNTY$ = {
			if = {
				limit = { has_county_modifier = civic_privilege_$PRIVILEGE$_2_county_modifier }
				save_scope_value_as = {
					name = burgher_grant_tier
					value = 3
				}
			}
			else_if = {
				limit = { has_county_modifier = civic_privilege_$PRIVILEGE$_1_county_modifier }
				save_scope_value_as = {
					name = burgher_grant_tier
					value = 2
				}
			}
			else = {
				save_scope_value_as = {
					name = burgher_grant_tier
					value = 1
				}
			}
		}

		burgher_remove_privilege_tiers_effect = {
			COUNTY = $COUNTY$
			PRIVILEGE = $PRIVILEGE$
		}

		$COUNTY$ = {
			if = {
				limit = { scope:burgher_grant_tier >= 3 }
				add_county_modifier = {
					modifier = civic_privilege_$PRIVILEGE$_3_county_modifier
				}
			}
			else_if = {
				limit = { scope:burgher_grant_tier >= 2 }
				add_county_modifier = {
					modifier = civic_privilege_$PRIVILEGE$_2_county_modifier
				}
			}
			else = {
				add_county_modifier = {
					modifier = civic_privilege_$PRIVILEGE$_1_county_modifier
				}
			}
		}

		burgher_advance_privilege_level_effect = { COUNTY = $COUNTY$ }
	}
}

burgher_provoke_county_factions_effect = {
	$COUNTY$ = {
		add_county_modifier = {
			modifier = burgher_broken_charter_county_modifier
			years = burgher_broken_charter_years_value
		}

		holder ?= { save_scope_as = burgher_revoke_liege }

		if = {
			limit = {
				any_title_joined_faction = {
					faction_is_type = burgher_autonomy_faction
				}
			}
			random_title_joined_faction = {
				limit = { faction_is_type = burgher_autonomy_faction }
				save_scope_as = burgher_revoke_faction
			}
		}
		else_if = {
			limit = {
				exists = scope:burgher_revoke_liege
				scope:burgher_revoke_liege = {
					any_targeting_faction = {
						faction_is_type = burgher_autonomy_faction
						$COUNTY$ = { can_title_join_faction = prev }
					}
				}
			}
			scope:burgher_revoke_liege = {
				random_targeting_faction = {
					limit = {
						faction_is_type = burgher_autonomy_faction
						$COUNTY$ = { can_title_join_faction = prev }
					}
					save_scope_as = burgher_revoke_faction
				}
			}
			title_join_faction = scope:burgher_revoke_faction
		}
		else_if = {
			limit = {
				exists = scope:burgher_revoke_liege
				can_title_create_faction = {
					type = burgher_autonomy_faction
					target = scope:burgher_revoke_liege
				}
			}
			title_create_faction = {
				type = burgher_autonomy_faction
				target = scope:burgher_revoke_liege
			}
			scope:burgher_revoke_liege = {
				random_targeting_faction = {
					limit = {
						faction_is_type = burgher_autonomy_faction
						any_faction_county_member = {
							this = $COUNTY$
						}
					}
					save_scope_as = burgher_revoke_faction
				}
			}
		}
	}

	if = {
		limit = { exists = scope:burgher_revoke_faction }
		scope:burgher_revoke_faction = {
			if = {
				limit = { NOT = { has_variable = burgher_grievance } }
				set_variable = {
					name = burgher_grievance
					value = 0
				}
			}
			change_variable = {
				name = burgher_grievance
				add = 1
			}
			add_faction_discontent = burgher_revoke_discontent_value
		}
	}
}

		}

				exists = scope:new_liege

league_trade_route_cohesion_effect = {
	if = {
		limit = {
			$ROUTE$.trade_route_owner ?= { is_confederation_member = yes }
		}
		$ROUTE$.trade_route_owner = {
			save_scope_as = league_route_owner
			confederation = { save_scope_as = league_route_league }
		}
	}
	else_if = {
		limit = {
			scope:trade_route_owner ?= { is_confederation_member = yes }
		}
		scope:trade_route_owner = {
			save_scope_as = league_route_owner
			confederation = { save_scope_as = league_route_league }
		}
	}

	if = {
		limit = { exists = scope:league_route_league }

		$ROUTE$ = {
			every_trade_market_on_route = {
				limit = { burgher_league_route_peer_market_trigger = yes }
				add_to_temporary_list = league_route_cohesion_markets
			}
		}

		if = {
			limit = {
				NOT = {
					any_in_list = {
						list = league_route_cohesion_markets
						count >= 1
					}
				}
				exists = scope:lost_trade_market
			}
			scope:lost_trade_market = {
				if = {
					limit = { burgher_league_route_peer_market_trigger = yes }
					add_to_temporary_list = league_route_cohesion_markets
				}
			}
		}

		if = {
			limit = {
				any_in_list = {
					list = league_route_cohesion_markets
					count >= 1
				}
			}
			$ROUTE$ = {
				set_variable = {
					name = league_route_peer_count
					value = 0
				}
			}
			every_in_list = {
				list = league_route_cohesion_markets
				scope:league_route_league = {
					change_cohesion = $AMOUNT$
				}
				$ROUTE$ = {
					change_variable = {
						name = league_route_peer_count
						add = 1
					}
				}
			}
		}
		else_if = {
			limit = {
				$ROUTE$ = {
					has_variable = league_route_peer_count
					var:league_route_peer_count > 0
				}
			}
			$ROUTE$ = {
				set_variable = {
					name = league_route_cohesion_left
					value = var:league_route_peer_count
				}
			}
			while = {
				limit = {
					$ROUTE$.var:league_route_cohesion_left > 0
				}
				scope:league_route_league = {
					change_cohesion = $AMOUNT$
				}
				$ROUTE$ = {
					change_variable = {
						name = league_route_cohesion_left
						subtract = 1
					}
				}
			}
			$ROUTE$ = { remove_variable = league_route_cohesion_left }
		}
	}
}
