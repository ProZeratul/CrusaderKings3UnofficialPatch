unop_save_adventurer_location_effect = {
	if = {
		limit = {
			exists = scope:adventurer
			NOT = { exists = scope:location }
		}
		scope:adventurer.location = {
			save_temporary_scope_as = location
		}
	}
}

#Unop Trigger physician treatments for the employer and their court; called from the court physician's on_court_position_received hook so every appointment route works, not just the paid hire event
unop_trigger_physician_treatments_effect = {
	hidden_effect = {
		if = {
			limit = {
				has_treatable_disease_trigger = yes
				has_recent_treatment_trigger = no
			}
			save_scope_as = sick_character
			trigger_event = {
				id = health.3101
				days = { physician_treatment_delay_ruler_min physician_treatment_delay_ruler_max }
			}
		}
		else_if = {
			limit = {
				has_recent_wound_treatment_trigger = no
				has_wounds_trigger = yes
			}
			save_scope_as = sick_character
			trigger_event = {
				id = health.4001
				days = { physician_treatment_delay_ruler_min physician_treatment_delay_ruler_max }
			}
		}
		every_courtier_or_guest = {
			if = {
				limit = {
					has_treatable_disease_trigger = yes
					has_recent_treatment_trigger = no
				}
				set_worst_disease_effect = yes
				save_scope_as = sick_character
				decide_who_picks_disease_treatment_effect = yes #Sends health.3101 to them or 3102 to liege (you)
			}
			else_if = {
				limit = {
					has_recent_wound_treatment_trigger = no
					has_wounds_trigger = yes
				}
				save_scope_as = sick_character
				decide_who_picks_wound_treatment_effect = yes
			}
		}
	}
}

#Unop: Implement the "Allure of the Steppe" (char_trade_yurt) arrival courtier: nomad_yurt_paiza_char_gain_arrival_lvl_N
#In the paiza employer (character) scope, needs scope:merchant_origin. Call it once per merchant arrival (mpo_decisions_events.2140)
unop_generate_yurt_paiza_char_effect = {
	save_scope_as = unop_yurt_char_employer
	#Create the character in the employer scope, not the domicile one: the templates use this.culture/faith/rite
	if = {
		limit = { domicile ?= { has_domicile_parameter = nomad_yurt_paiza_char_gain_arrival_lvl_1 } }
		create_character = {
			location = scope:unop_yurt_char_employer.location
			culture = scope:merchant_origin.county.culture
			faith = scope:merchant_origin.county.faith
			rite = scope:merchant_origin.county.rite
			gender_female_chance = 50
			template = peasant_character
			after_creation = {
				add_trait = lifestyle_traveler
			}
			save_scope_as = yurt_merchant_char
		}
	}
	else_if = {
		limit = { domicile ?= { has_domicile_parameter = nomad_yurt_paiza_char_gain_arrival_lvl_2 } }
		create_character = {
			location = scope:unop_yurt_char_employer.location
			culture = scope:merchant_origin.county.culture
			faith = scope:merchant_origin.county.faith
			rite = scope:merchant_origin.county.rite
			gender_female_chance = 50
			template = merchant_template
			after_creation = {
				add_trait = lifestyle_traveler
			}
			save_scope_as = yurt_merchant_char
		}
	}
	else_if = {
		limit = { domicile ?= { has_domicile_parameter = nomad_yurt_paiza_char_gain_arrival_lvl_3 } }
		create_character = {
			location = scope:unop_yurt_char_employer.location
			culture = scope:merchant_origin.county.culture
			faith = scope:merchant_origin.county.faith
			rite = scope:merchant_origin.county.rite
			gender_female_chance = 50
			template = merchant_template
			after_creation = {
				add_random_skill_point_effect = yes
				add_random_skill_point_effect = yes
				add_random_skill_point_effect = yes
				add_random_skill_point_effect = yes
				add_random_skill_point_effect = yes
				add_random_skill_point_effect = yes
				random_list = {
					1 = { add_trait = avaricious }
					1 = { add_trait = diplomat }
					1 = { add_trait = overseer }
				}
				add_trait = lifestyle_traveler
			}
			save_scope_as = yurt_merchant_char
		}
	}
	if = {
		limit = { exists = scope:yurt_merchant_char }
		hidden_effect = { add_courtier = scope:yurt_merchant_char }
	}
}

#Unop Converts every county in a ruler's domain still in $OLD_FAITH$ to $RITE$, e.g. after a pagan reform moved the ruler into a new faith
unop_convert_old_faith_domain_counties_effect = {
	every_held_title = {
		limit = {
			tier = tier_county
			faith = $OLD_FAITH$
		}
		set_county_faith = $RITE$.faith
		set_county_rite = $RITE$
	}
}
