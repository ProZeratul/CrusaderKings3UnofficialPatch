#Unop Trigger physician treatments for the employer and their court; called from the court physician's on_court_position_received hook so every appointment route works, not just the paid hire event
unop_trigger_physician_treatments_effect = {
	hidden_effect = {
		if = {
			limit = {
				has_treatable_disease_trigger = yes
				has_recent_treatment_trigger = no
			}
			save_scope_as = sick_character
			trigger_event = {
				id = health.3101
				days = { physician_treatment_delay_ruler_min physician_treatment_delay_ruler_max }
			}
		}
		else_if = {
			limit = {
				has_recent_wound_treatment_trigger = no
				has_wounds_trigger = yes
			}
			save_scope_as = sick_character
			trigger_event = {
				id = health.4001
				days = { physician_treatment_delay_ruler_min physician_treatment_delay_ruler_max }
			}
		}
		every_courtier_or_guest = {
			if = {
				limit = {
					has_treatable_disease_trigger = yes
					has_recent_treatment_trigger = no
				}
				set_worst_disease_effect = yes
				save_scope_as = sick_character
				decide_who_picks_disease_treatment_effect = yes #Sends health.3101 to them or 3102 to liege (you)
			}
			else_if = {
				limit = {
					has_recent_wound_treatment_trigger = no
					has_wounds_trigger = yes
				}
				save_scope_as = sick_character
				decide_who_picks_wound_treatment_effect = yes
			}
		}
	}
}

#Unop Converts every county in a ruler's domain still in $OLD_FAITH$ to $RITE$, e.g. after a pagan reform moved the ruler into a new faith

#Unop Governors with no valid appointment candidate pass their titles to the top liege, not to the engine's default fallback (child, parent, etc.)
unop_update_governor_fallback_succession_effect = {
	if = {
		limit = {
			is_governor = yes
			OR = {
				has_character_flag = unop_governor_fallback_succession
				NOR = { # Don't overwrite a fallback set elsewhere, e.g. a church succession promise
					exists = fallback_succession_title
					exists = fallback_succession_character
				}
			}
		}
		set_fallback_succession_title = top_liege.primary_title
		add_character_flag = unop_governor_fallback_succession
	}
	else_if = {
		limit = {
			is_governor = no
			is_ruler = yes # Only rulers have a succession to clear, a former governor without titles keeps the flag until ruling again
			has_character_flag = unop_governor_fallback_succession
		}
		clear_fallback_succession = yes
		remove_character_flag = unop_governor_fallback_succession
	}
}

#Unop Rolls the injury of maimed_in_battle_effect, with the same weights, into scope:unop_maimed_in_battle_injury
#Roll it outside a toast and apply it inside with unop_apply_maimed_in_battle_effect, so that the toast names the injury applied, not one of a separate roll
unop_roll_maimed_in_battle_effect = {
	random_list = {
		4 = {
			trigger = {
				NOT = {
					has_trait = one_legged
				}
			}
			save_scope_value_as = {
				name = unop_maimed_in_battle_injury
				value = flag:one_legged
			}
		}
		2 = {
			trigger = {
				NOT = {
					has_trait = disfigured
				}
			}
			save_scope_value_as = {
				name = unop_maimed_in_battle_injury
				value = flag:disfigured
			}
		}
		4 = {
			trigger = {
				NOT = {
					has_trait = one_eyed
				}
			}
			save_scope_value_as = {
				name = unop_maimed_in_battle_injury
				value = flag:one_eyed
			}
		}
		4 = {
			trigger = {
				NOT = {
					has_trait = maimed
				}
			}
			save_scope_value_as = {
				name = unop_maimed_in_battle_injury
				value = flag:maimed
			}
		}
	}
}

#Unop Applies the injury rolled by unop_roll_maimed_in_battle_effect, as maimed_in_battle_effect does
unop_apply_maimed_in_battle_effect = {
	if = {
		limit = { scope:unop_maimed_in_battle_injury ?= flag:one_legged }
		add_trait_force_tooltip = one_legged
		increase_wounds_effect = { REASON = fight }
	}
	else_if = {
		limit = { scope:unop_maimed_in_battle_injury ?= flag:disfigured }
		add_trait_force_tooltip = disfigured
		increase_wounds_effect = { REASON = fight }
		epilepsy_brain_trauma_risk_effect = { CHANCE = 5 }
	}
	else_if = {
		limit = { scope:unop_maimed_in_battle_injury ?= flag:one_eyed }
		add_trait_force_tooltip = one_eyed
		increase_wounds_effect = { REASON = fight }
		epilepsy_brain_trauma_risk_effect = { CHANCE = 5 }
	}
	else_if = {
		limit = { scope:unop_maimed_in_battle_injury ?= flag:maimed }
		apply_maimed_trait_and_modifier_effect = yes
		epilepsy_brain_trauma_risk_effect = { CHANCE = 5 }
	}
}

#Unop A character who gives up all titles is briefly non-playable, and the engine then drops the tenets and doctrines they learned and their personal tenets
# Call unop_save_tenet_knowledge_effect before the titles are transferred, and unop_restore_tenet_knowledge_effect once the character holds a title again, in the same effect
unop_save_tenet_knowledge_effect = {
	every_character_tenet = {
		knowledge_filter = known
		add_to_list = unop_known_tenets
	}
	every_character_doctrine = {
		knowledge_filter = known
		add_to_list = unop_known_doctrines
	}
	every_personal_tenet = {
		add_to_list = unop_personal_tenets
	}
}

unop_restore_tenet_knowledge_effect = {
	save_scope_as = unop_tenet_knower
	hidden_effect = {
		every_in_list = {
			list = unop_known_tenets
			save_scope_as = unop_tenet
			scope:unop_tenet_knower = {
				if = {
					limit = {
						NOT = { knows_tenet = scope:unop_tenet }
					}
					add_known_tenet = scope:unop_tenet
				}
			}
		}
		every_in_list = {
			list = unop_known_doctrines
			save_scope_as = unop_doctrine
			scope:unop_tenet_knower = {
				if = {
					limit = {
						NOT = { knows_doctrine = scope:unop_doctrine }
					}
					add_known_doctrine = scope:unop_doctrine
				}
			}
		}
		every_in_list = {
			list = unop_personal_tenets
			save_scope_as = unop_tenet
			scope:unop_tenet_knower = {
				if = {
					limit = {
						NOT = { has_personal_tenet = scope:unop_tenet }
					}
					force_set_personal_tenet = scope:unop_tenet
				}
			}
		}
	}
}
