# Effect for founding and becoming a republic
# Use on founder or character scope
# NOTE: Order of operation is important here
gdp_found_a_republic_effect = {
	## Pre-setup
	add_character_flag = gdp_founding_republic
	# Save scopes for later use
	save_scope_as = republican_founder
	if = {
		limit = {
			is_landed = yes
		}
		primary_title ?= { save_scope_as = old_title }
	}
	if = {
		limit = {
			OR = {
				NOT = { exists = domicile }
				domicile ?= {
					NOT = { is_domicile_type = merchant_company }
				}
			}
		}
		save_temporary_scope_as = create_merchant_title
	}

	# Let's save your current base income, to use as a base for how big your starting treasury will be
	save_scope_value_as = {
		name = income_treasury_conversion
		value = monthly_character_income
	}

	## Base setup
	# 1. Are we already landed and do we have a higher tier title?
	if = {
		limit = {
			exists = scope:old_title
			scope:old_title.tier > tier_county
		}
		custom_tooltip = {
			text = found_a_republic_new_title_desc

			# Then create a new republican title of the corresponding tier
			random_list = {
				10 = {
					trigger = {
						scope:old_title.tier = tier_kingdom
					}
					create_dynamic_title = {
						type = title_type_republic
						tier = kingdom
						holder = scope:republican_founder
						name = NEW_REPUBLIC_TITLE_NAME
						adj = NEW_REPUBLIC_TITLE_ADJ
						save_scope_as = new_republic_title
					}
				}
				10 = {
					trigger = {
						scope:old_title.tier = tier_duchy
					}
					create_dynamic_title = {
						type = title_type_republic
						tier = duchy
						holder = scope:republican_founder
						name = NEW_REPUBLIC_TITLE_NAME
						adj = NEW_REPUBLIC_TITLE_ADJ
						save_scope_as = new_republic_title
					}
				}
			}

			# Set up the new title
			if = {
				limit = {
					exists = scope:new_republic_title
				}
				scope:new_republic_title ?= {
					set_coa = scope:republican_founder.capital_county
					set_color_from_title = scope:republican_founder.capital_county
					set_capital_county = scope:republican_founder.capital_county
					set_no_automatic_claims = yes
				}
				set_primary_title_to = scope:new_republic_title
			}
		}
	}
	# Fallback: county-only purchases and landless founders have no higher title, so we still make a duchy-tier republic title
	else_if = {
		limit = {
			NOT = {
				exists = scope:new_republic_title
			}
		}
		hidden_effect = {
			create_dynamic_title = {
				type = title_type_republic
				tier = duchy
				holder = scope:republican_founder
				name = NEW_REPUBLIC_TITLE_NAME
				adj = NEW_REPUBLIC_TITLE_ADJ
				save_scope_as = new_republic_title
			}
			if = {
				limit = { exists = scope:new_republic_title }
				scope:new_republic_title ?= {
					set_coa = scope:republican_founder.capital_county
					set_color_from_title = scope:republican_founder.capital_county
					set_capital_county = scope:republican_founder.capital_county
					set_no_automatic_claims = yes
				}
				set_primary_title_to = scope:new_republic_title
			}
		}
	}

	# 2. Are we house head? If not, form a cadet branch
	if = {
		limit = {
			exists = house.house_head
			house.house_head != scope:republican_founder
		}
		custom_tooltip = {
			text = found_a_republic_form_cadet_branch_desc
			create_cadet_branch = {}
		}
	}

	# 3. Become a republic
	change_government = republic_government

	# 4. Create a Merchant/Patrician family title
	if = {
		limit = {
			exists = scope:create_merchant_title
			NOT = {
				domicile ?= {
					is_domicile_type = merchant_company
				}
			}
		}
		# Merchant titles also count as noble family titles and do not need to be destroyed, but real noble family titles do
		# We destroy them here, or the new merchant title is not given
		if = {
			limit = {
				any_held_title = {
					is_noble_family_title = yes
				}
			}
			every_held_title = {
				limit = {
					is_noble_family_title = yes
				}
				root = {
					destroy_title = prev
				}
			}
		}
		custom_tooltip = {
			text = found_a_republic_new_merchant_title_desc
			create_dynamic_title = {
				name = NEW_PATRICIAN_TITLE_NAME
				tier = county
				type = title_type_merchant_family
				holder = scope:republican_founder
				article = DEFAULT_TITLE_NAME_ARTICLE
				save_scope_as = new_patrician_title
			}
			scope:new_patrician_title ?= {
				add_title_law = noble_family_succession_law
			}
		}
	}
	else_if = {
		limit = {
			domicile ?= {
				is_domicile_type = merchant_company
			}
		}
		custom_tooltip = {
			text = found_a_republic_new_merchant_title_desc
			gdp_republic_restore_family_succession_law_effect = yes
		}
	}

	# 5. Destroy our old titles
	if = {
		limit = {
			any_held_title = {
				tier > tier_county
				trigger_if = {
					limit = {
						exists = scope:new_republic_title
					}
					this != scope:new_republic_title
				}
			}
		}
		every_held_title = {
			limit = {
				tier > tier_county
				trigger_if = {
					limit = {
						exists = scope:new_republic_title
					}
					this != scope:new_republic_title
				}
			}
			scope:republican_founder = {
				destroy_title = prev
			}
		}
	}

	# 6. Extra patricians only at duchy tier, matching republic_tier_duchy
	if = {
		limit = {
			OR = {
				scope:old_title ?= { tier >= tier_duchy }
				scope:new_republic_title ?= { tier >= tier_duchy }
			}
		}
		custom_tooltip = {
			text = found_a_republic_joining_patricians_desc
			new_republic_population_effect = yes
		}
	}

	## Final setup
	# Add a founder bonus
	add_character_modifier = {
		modifier = gdp_founded_republic_modifier
	}

	# Add starting treasury, so it is never empty
	add_treasury = {
		value = scope:income_treasury_conversion
		round = yes
		multiply = 36 # 3 years worth of income
		multiply = { # Scaled further with tier
			value = primary_title.tier
			subtract = 1 # So that count tier is 1
		}
		min = 100
	}
	# GDP music cue, only plays for the local player
	scope:republican_founder = { play_music_cue = mx_cue_the_rise_of_a_nation }
	remove_character_flag = gdp_founding_republic

	# Move Domicile
	domicile ?= {
		if = {
			limit = { domicile_location != scope:republican_founder.capital_province }
			move_domicile = scope:republican_founder.capital_province
		}
	}

	# Warning of elector loss
	gdp_hre_elector_loss_warning_effect = yes

	# Make sure we don't found a republic multiple times
	add_character_flag = { flag = gdp_recently_founded_republic days = 1 }
}

# Starts the scoped republic head and every patrician family of their republic at level 2
# Call right after gdp_found_a_republic_effect for republics the game spawns, not for rulers choosing to become one
gdp_republic_patricians_start_at_level_2_effect = {
	hidden_effect = {
		domicile ?= {
			if = {
				limit = { is_domicile_type = merchant_company }
				gdp_patrician_domicile_level_2_effect = yes
			}
		}
		every_vassal = {
			limit = {
				government_has_flag = government_is_republic
				domicile ?= { is_domicile_type = merchant_company }
			}
			domicile = { gdp_patrician_domicile_level_2_effect = yes }
		}
	}
}

# We only fill free slots, so a company that already has buildings keeps them
gdp_patrician_domicile_level_2_internal_effect = {
	if = {
		limit = {
			NOT = { has_domicile_building_or_higher = merchant_company_$BUILDING$_01 }
			domicile_building_has_free_internal_slot = merchant_company_$PARENT$_02
		}
		add_domicile_building = merchant_company_$BUILDING$_01
	}
	if = {
		limit = { has_domicile_building = merchant_company_$BUILDING$_01 }
		add_domicile_building = merchant_company_$BUILDING$_02
	}
}

# New independent republic in $COUNTY$, enrolled in $LEAGUE_HEAD$'s league after gdp_found_a_republic_effect creates the primary title
gdp_trade_league_proclaim_commune_effect = {
	$COUNTY$ = { save_scope_as = commune_county }
	$LEAGUE_HEAD$ = { save_scope_as = commune_patron }

	# Tooltip is written by hand: the new ruler does not exist in preview
	custom_tooltip = league_commune_cb_proclaimed_tt
	custom_tooltip = league_commune_cb_joins_league_tt

	hidden_effect = {
		create_character = {
			location = scope:commune_county.title_province
			template = burgher_faction_leader_template
			faith = scope:commune_county.faith
			culture = scope:commune_county.culture
			gender_female_chance = {
				if = {
					limit = {
						scope:commune_county.rite = {
							rite_has_doctrine = doctrine_gender_male_dominated
						}
					}
					add = 0
				}
				else_if = {
					limit = {
						scope:commune_county.rite = {
							rite_has_doctrine = doctrine_gender_female_dominated
						}
					}
					add = 100
				}
				else = {
					add = 50
				}
			}
			save_scope_as = commune_patrician
		}

		if = {
			limit = { exists = scope:commune_patrician }

			scope:commune_patrician = {
				set_character_rite = scope:commune_county.rite
			}

			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = commune_change
				add_claim_on_loss = yes
			}

			scope:commune_county = {
				change_title_holder = {
					holder = scope:commune_patrician
					change = scope:commune_change
				}
			}

			# Grant independence: create_character leaves them in the old owner's court
			scope:commune_patrician = {
				becomes_independent = { change = scope:commune_change }
			}

			resolve_title_and_vassal_change = scope:commune_change

			burgher_found_city_republic_effect = { RULER = scope:commune_patrician }

			scope:commune_county = {
				if = {
					limit = { NOT = { has_variable = civic_league_charter } }
					set_variable = civic_league_charter
				}
			}

			scope:commune_patron.confederation ?= {
				add_confederation_member_title = scope:commune_patrician.primary_title
			}

			scope:commune_patrician = {
				add_opinion = {
					modifier = league_commune_founder_opinion
					target = scope:commune_patron
				}
			}
		}
	}
}

# Complete monopoly in $COUNTY$'s market for $LEAGUE_HEAD$'s league
gdp_trade_league_enforce_monopoly_effect = {
	$COUNTY$ = { save_scope_as = monopoly_county }
	$LEAGUE_HEAD$ = { save_scope_as = monopoly_enforcer }
	$LEAGUE_HEAD$.confederation = { save_scope_as = trade_league_monopoly }
	$COUNTY$.title_province.barony.trade_market = { save_scope_as = trade_market }

	custom_tooltip = request_monopoly_trade_league_tt_complete_normal

	hidden_effect = {
		create_trade_monopoly = {
			owning_confederation = scope:trade_league_monopoly
			trade_market = scope:trade_market
			type = normal
			is_complete = yes
		}
		scope:monopoly_enforcer = {
			save_scope_as = actor
			save_scope_as = monopoly_creator
		}
		scope:trade_market = {
			save_scope_as = monopolized_market
			every_character_in_market = {
				limit = {
					is_ai = no
					NOR = {
						this = scope:monopoly_enforcer
						is_member_of_confederation = scope:trade_league_monopoly
						any_liege_or_above = {
							is_member_of_confederation = scope:trade_league_monopoly
						}
					}
				}
				send_interface_toast = {
					title = trade_monopoly_created_against_you_toast_title
					type = msg_trade_monopoly
					left_icon = scope:monopoly_enforcer
					desc = trade_monopoly_created_against_you_complete_toast_desc
				}
			}
		}
	}
}

# Contested assets for the trade war victory effect
gdp_league_collect_contested_trade_assets_effect = {
	every_owned_trade_post_district = {
		limit = {
			district_province = {
				NOT = { is_in_list = surrendered_trade_posts }
				barony.trade_market ?= {
					gdp_league_contested_market_trigger = {
						LEAGUE = scope:victorious_league
						RIVAL_LEAGUE = scope:defeated_league
					}
				}
			}
		}
		district_province = { add_to_list = surrendered_trade_posts }
	}

	every_trade_monopoly_for_character = {
		limit = {
			NOT = { is_in_list = surrendered_monopolies }
			monopoly_trade_market ?= {
				gdp_league_contested_market_trigger = {
					LEAGUE = scope:victorious_league
					RIVAL_LEAGUE = scope:defeated_league
				}
			}
		}
		add_to_list = surrendered_monopolies
	}

	# Republic and city titles hold monopolies in their own right
	every_held_title = {
		every_trade_monopoly_for_title = {
			limit = {
				NOT = { is_in_list = surrendered_monopolies }
				monopoly_trade_market ?= {
					gdp_league_contested_market_trigger = {
						LEAGUE = scope:victorious_league
						RIVAL_LEAGUE = scope:defeated_league
					}
				}
			}
			add_to_list = surrendered_monopolies
		}
	}
}

# Strip $RIVAL_LEAGUE_HEAD$'s league from markets contested with $LEAGUE_HEAD$
gdp_trade_league_win_trade_war_effect = {
	$LEAGUE_HEAD$.confederation = { save_scope_as = victorious_league }
	$RIVAL_LEAGUE_HEAD$.confederation = { save_scope_as = defeated_league }

	custom_tooltip = league_trade_war_posts_tt
	custom_tooltip = league_trade_war_monopolies_tt

	hidden_effect = {
		scope:defeated_league = {
			every_confederation_member = {
				gdp_league_collect_contested_trade_assets_effect = yes

				# Patrician families hold posts and monopolies on their republic's behalf
				every_vassal_or_below = {
					gdp_league_collect_contested_trade_assets_effect = yes
				}
			}

			every_trade_monopoly_for_confederation = {
				limit = {
					monopoly_trade_market ?= {
						gdp_league_contested_market_trigger = {
							LEAGUE = scope:victorious_league
							RIVAL_LEAGUE = scope:defeated_league
						}
					}
				}
				add_to_list = surrendered_monopolies
			}
		}

		# Remove after the loops, or iteration stops early
		every_in_list = {
			list = surrendered_trade_posts
			remove_district_type = trade_post_location_district
		}
		every_in_list = {
			list = surrendered_monopolies
			destroy_trade_monopoly = yes
		}
	}
}

# Adds the trade league the scoped character trades on behalf of to $LIST$
gdp_league_add_affiliated_league_to_list_effect = {
	if = {
		limit = { gdp_is_trade_league_member_trigger = yes }
		confederation = {
			if = {
				limit = { NOT = { is_in_list = $LIST$ } }
				add_to_list = $LIST$
			}
		}
	}
	else_if = {
		limit = {
			any_liege_or_above = { gdp_is_trade_league_member_trigger = yes }
		}
		# Nearest league-member liege
		ordered_liege_or_above = {
			limit = { gdp_is_trade_league_member_trigger = yes }
			order_by = {
				value = 0
				subtract = highest_held_title_tier
			}
			confederation = {
				if = {
					limit = { NOT = { is_in_list = $LIST$ } }
					add_to_list = $LIST$
				}
			}
		}
	}
}

# Adds the trade league behind the scoped monopoly to $LIST$
gdp_monopoly_add_league_to_list_effect = {
	if = {
		limit = {
			confederation_owner ?= {
				confederation_type = confederation_type:trade_league
			}
		}
		confederation_owner = {
			if = {
				limit = { NOT = { is_in_list = $LIST$ } }
				add_to_list = $LIST$
			}
		}
	}
	else_if = {
		limit = { exists = character_owner }
		character_owner = { gdp_league_add_affiliated_league_to_list_effect = { LIST = $LIST$ } }
	}
	else_if = {
		limit = { exists = title_owner.holder }
		title_owner.holder = { gdp_league_add_affiliated_league_to_list_effect = { LIST = $LIST$ } }
	}
}

# Strip rival leagues from $MARKET$ within $REALM_HEAD$'s realm
gdp_league_expel_rivals_from_market_effect = {
	$LEAGUE_HEAD$ = { save_scope_as = expulsion_demander }
	$LEAGUE_HEAD$.confederation = { save_scope_as = expelling_league }
	$REALM_HEAD$ = { save_scope_as = expulsion_host }
	$MARKET$ = { save_scope_as = expulsion_market }

	custom_tooltip = league_expulsion_posts_tt
	custom_tooltip = league_expulsion_cohesion_shift_tt

	hidden_effect = {
		# Collect before removing
		scope:expulsion_host = {
			every_realm_province = {
				limit = {
					exists = trade_post_district
					barony.trade_market ?= scope:expulsion_market
					trade_post_owner = {
						gdp_league_foreign_competitor_trigger = {
							LEAGUE = scope:expelling_league
							REALM_HEAD = scope:expulsion_host
						}
					}
				}
				add_to_list = expelled_trade_posts
				trade_post_owner = {
					gdp_league_add_affiliated_league_to_list_effect = { LIST = expelled_leagues }
				}
			}
		}

		scope:expulsion_market = {
			every_trade_monopoly_for_market = {
				limit = {
					gdp_monopoly_held_for_rival_league_trigger = {
						LEAGUE = scope:expelling_league
						REALM_HEAD = scope:expulsion_host
					}
				}
				add_to_list = expelled_monopolies
				gdp_monopoly_add_league_to_list_effect = { LIST = expelled_leagues }
			}
		}

		every_in_list = {
			list = expelled_trade_posts
			remove_district_type = trade_post_location_district
		}
		every_in_list = {
			list = expelled_monopolies
			destroy_trade_monopoly = yes
		}

		every_in_list = {
			list = expelled_leagues
			change_cohesion = gdp_league_expulsion_cohesion_value

			every_confederation_member = {
				add_opinion = {
					modifier = league_expelled_from_market_opinion
					target = scope:expulsion_demander
				}
				add_opinion = {
					modifier = league_expelled_from_market_opinion
					target = scope:expulsion_host
				}
				if = {
					limit = { is_ai = no }
					send_interface_message = {
						type = msg_trade_monopoly
						title = league_expelled_from_market_message_title
						desc = league_expelled_from_market_message_desc
						left_icon = scope:expulsion_demander
					}
				}
			}
		}
	}
}

# Saves this character's republic head as scope:patrician_liege, if they have one

		limit = {
			any_liege_or_above = { gdp_is_assembly_head_trigger = yes }

	}
}

# Sets up a Ruler Designer character as a new patrician family of an existing republic.
gdp_create_patrician_family_effect = {
	save_scope_as = new_patrician
	save_scope_as = republican_founder # NEW_PATRICIAN_TITLE_NAME reads the house off this scope

	# The family belongs to the republic that holds the county it was created in
	location.county.holder ?= { gdp_save_patrician_liege_effect = yes }

	# House titles require every potential successor to share the holder's house.
	house = { save_scope_as = patrician_title_house }
	every_close_family_member = {
		limit = {
			is_alive = yes
			NOT = { exists = house }
		}
		set_house = scope:patrician_title_house
	}

	create_dynamic_title = {
		name = NEW_PATRICIAN_TITLE_NAME
		tier = county
		type = title_type_merchant_family
		holder = scope:new_patrician
		article = DEFAULT_TITLE_NAME_ARTICLE
		save_scope_as = new_patrician_title
	}
	scope:new_patrician_title ?= {
		add_title_law = noble_family_succession_law
		set_coa = scope:new_patrician.house
	}

	# Join the republic
	if = {
		limit = {
			NOT = { liege ?= scope:patrician_liege }
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = patrician_vassal_change
		}
		change_liege = {
			liege = scope:patrician_liege
			change = scope:patrician_vassal_change
		}
		resolve_title_and_vassal_change = scope:patrician_vassal_change
	}
	if = {
		limit = {
			NOT = { has_government = republic_government }
		}
		change_government = republic_government
	}

	# Join the assembly now so the succession recalculation at game start builds the electors with this family
	scope:patrician_liege ?= { update_assembly = yes }

	# Republicans that hold land use treasury, give them a small starting pool
	if = {
		limit = {
			has_treasury = yes
		}
		add_treasury = 100
	}

	# The merchant company domicile comes with the government; put it where the family actually lives
	domicile ?= {
		if = {
			limit = { domicile_location != scope:new_patrician.location }
			move_domicile = scope:new_patrician.location
		}
	}
}

gdp_convert_tribal_holdings_for_republic_effect = {
	capital_province ?= { save_scope_as = republic_capital_province }

	every_realm_province = {
		limit = {
			has_holding_type = tribal_holding
			NOT = { is_in_list = republic_conversion }
		}
		add_to_list = republic_conversion
	}
	every_vassal_or_below = {
		every_realm_province = {
			limit = {
				has_holding_type = tribal_holding
				NOT = { is_in_list = republic_conversion }
			}
			add_to_list = republic_conversion
		}
	}
	every_in_list = {
		list = republic_conversion
		if = {
			limit = {
				exists = scope:republic_capital_province
				this = scope:republic_capital_province
			}
			holding_convert_tribal_keeping_buildings_effect = { HOLDING = city_holding }
		}
		else = {
			holding_convert_tribal_keeping_buildings_effect = { HOLDING = castle_holding }
		}
		remove_from_list = republic_conversion
	}
}

gdp_feudalize_tribal_vassals_for_republic_effect = {
	every_vassal_or_below = {
		limit = { government_has_flag = government_is_tribal }

		}
		else = {
			change_government = feudal_government
		}

			limit = {
				NOT = { has_realm_law = crown_authority_0 }
			}
			add_realm_law_skip_effects = crown_authority_0
		}
	}
}

# Rank up Republic - Use on the ruler of the republic
gdp_rank_up_republic_effect = {
	save_scope_as = republican_founder
	primary_title = {
		save_scope_as = old_republic_title
		title_confederation ?= {
			if = {
				limit = { confederation_type = confederation_type:trade_league }
				save_scope_as = trade_league
				leading_title = { save_scope_as = old_league_leader }
			}
		}
	}

	# Create a title of your next tier
	if = {
		limit = {
			primary_title.tier = tier_kingdom
		}
		create_dynamic_title = {
			type = title_type_republic
			tier = empire
			holder = scope:republican_founder
			name = NEW_REPUBLIC_TITLE_TIER_NAME
			adj = NEW_REPUBLIC_TITLE_TIER_ADJ
			save_scope_as = new_republic_title
		}
		custom_tooltip = republic_rank_up_empire_title_desc
	}
	else_if = {
		limit = {
			primary_title.tier = tier_duchy
		}
		create_dynamic_title = {
			type = title_type_republic
			tier = kingdom
			holder = scope:republican_founder
			name = NEW_REPUBLIC_TITLE_TIER_NAME
			adj = NEW_REPUBLIC_TITLE_TIER_ADJ
			save_scope_as = new_republic_title
		}
		custom_tooltip = republic_rank_up_kingdom_title_desc
	}
	else_if = { # I.e. county
		limit = {
			primary_title.tier = tier_county
		}
		create_dynamic_title = {
			type = title_type_republic
			tier = duchy
			holder = scope:republican_founder
			name = NEW_REPUBLIC_TITLE_TIER_NAME
			adj = NEW_REPUBLIC_TITLE_TIER_ADJ
			save_scope_as = new_republic_title
		}
		custom_tooltip = republic_rank_up_duchy_title_desc
	}

	# We'll hide the following effects for clarity - the important part above use already custom tooltips
	hidden_effect = {
		if = {
			limit = { exists = scope:new_republic_title }
			# Set up the new title
			scope:new_republic_title ?= {
				# CoA and color
				set_coa = scope:old_republic_title
				set_color_from_title = scope:old_republic_title
				# Copy old title history
				copy_title_history = scope:old_republic_title
				# Set capital
				set_capital_county = scope:old_republic_title.title_capital_county
				set_no_automatic_claims = yes
			}
			# Move Title MaAs to new title
			if = {
				limit = {
					scope:old_republic_title = {
						any_title_maa_regiment = { }
					}
				}
				scope:old_republic_title = { transfer_title_maa_ownership = scope:new_republic_title }
			}
			# Update Trade League member/leader if relevant
			scope:trade_league ?= {
				# For blocking on_member_join/leave effects
				add_to_variable_list = {
					name = rank_change_members
					target = scope:republican_founder
				}
				add_to_variable_list = {
					name = rank_change_titles
					target = scope:old_republic_title
				}
				add_to_variable_list = {
					name = rank_change_titles
					target = scope:new_republic_title
				}
				# Must remove first or code blocks holder from 'joining' again
				remove_confederation_member_title = scope:old_republic_title
				add_confederation_member_title = scope:new_republic_title
				if = {
					limit = { scope:old_league_leader ?= scope:old_republic_title }
					scope:republican_founder = { save_scope_as = league_creator }
					set_leading_title = scope:new_republic_title
					if = { # For safety, as title member changes can mess things up
						limit = { exists = var:is_hanseatic_league }
						set_coa = k_hansa
						set_confederation_name = HANSEATIC_LEAGUE_NAME
					}
					else = {
						set_coa = scope:new_republic_title
						set_confederation_name = TRADE_LEAGUE_BASE_NAME
					}
				}
				clear_variable_list = rank_change_members
				clear_variable_list = rank_change_titles
			}
			# Transfer existing monopolies to the new title
			if = {
				limit = {
					scope:old_republic_title = {
						title_has_trade_monopoly = yes
					}
				}
				scope:old_republic_title = {
					every_trade_monopoly_for_title = {
						# Set up what we need for creating a new monopoly
						# Trade Good
						if = {
							limit = { is_complete_monopoly = no }
							monopoly_trade_good_category = { save_scope_as = trade_good_category_type }
							scope:republican_founder = {
								set_variable = { name = new_monopoly_trade_good_category_type value = scope:trade_good_category_type }
							}
						}
						# Complete Monopoly
						else = {
							scope:republican_founder = {
								set_variable = { name = is_complete_monopoly value = yes }
							}
						}
						# Exploitative Monopoly
						if = {
							limit = {
								monopoly_type = exploitative
							}
							scope:republican_founder = {
								set_variable = { name = is_exploitative_monopoly value = yes }
							}
						}
						# Trade Market
						monopoly_trade_market = {
							save_scope_as = trade_market
							scope:republican_founder = {
								set_variable = { name = new_monopoly_trade_market value = scope:trade_market }
							}
						}

						# First, destroy the old monopoly
						destroy_trade_monopoly = yes

						# Then create the new monopoly
						scope:republican_founder = {
							if = {
								limit = {
									exists = var:is_complete_monopoly
									exists = var:is_exploitative_monopoly
								}
								create_trade_monopoly = {
									owning_title = scope:new_republic_title
									trade_market = var:new_monopoly_trade_market
									type = exploitative
									is_complete = yes
								}
							}
							else_if = {
								limit = {
									exists = var:is_complete_monopoly
								}
								create_trade_monopoly = {
									owning_title = scope:new_republic_title
									trade_market = var:new_monopoly_trade_market
									type = normal
									is_complete = yes
								}
							}
							else_if = {
								limit = {
									exists = var:is_exploitative_monopoly
								}
								create_trade_monopoly = {
									owning_title = scope:new_republic_title
									trade_market = var:new_monopoly_trade_market
									trade_good_category = var:new_monopoly_trade_good_category_type
									type = exploitative
								}
							}
							else = {
								create_trade_monopoly = {
									owning_title = scope:new_republic_title
									trade_market = var:new_monopoly_trade_market
									trade_good_category = var:new_monopoly_trade_good_category_type
									type = normal
								}
							}
							scope:new_republic_title = {
								random_trade_monopoly_for_title = {
									limit = {
										NOT = { is_in_list = new_title_monopolies }
									}
									save_scope_as = new_monopoly_scope
									if = {
										limit = { is_complete_monopoly = yes }
										set_trade_monopoly_name = new_complete_title_monopoly_name
									}
									else = {
										set_trade_monopoly_name = new_good_title_monopoly_name
									}
									add_to_list = new_title_monopolies
								}
							}

							# Finally, clear the variables
							remove_variable = new_monopoly_trade_good_category_type
							remove_variable = new_monopoly_trade_market
							remove_variable = is_complete_monopoly
							remove_variable = is_exploitative_monopoly
						}
					}
				}
			}
			# Destroy old title if needed
			if = {
				limit = { scope:old_republic_title.tier > tier_county }
				scope:republican_founder = { destroy_title = scope:old_republic_title }
			}
		}
	}
}

	save_scope_value_as = { # Note: the ruler is not in the vassal count, so the real number is usually one higher
		name = number_of_expected_families
		value = {
			value = 0
			# Tier
			if = {
				limit = { highest_held_title_tier >= tier_empire }
				add = 9
			}
			else_if = {
				limit = { highest_held_title_tier = tier_kingdom }
				add = 7
			}
			else = {
				add = 4
			}
			# Realm size
			if = {
				limit = { realm_size >= 50 }
				add = 3
			}
			else_if = {
				limit = { realm_size >= 25 }
				add = 2
			}
			else_if = {
				limit = { realm_size >= 10 }
				add = 1
			}
		}
	}

	# Let's start by checking if we have enough families for our tier
	if = {
		limit = {
			scope:republican = {
				gdp_is_assembly_head_trigger = yes
				highest_held_title_tier >= tier_duchy
				any_vassal = {
					government_has_flag = government_is_republic
					exists = domicile
					domicile = {
						is_domicile_type = merchant_company
					}
					count < scope:number_of_expected_families
				}
			}
		}

		# We have too few families, let's introduce one
		hidden_effect = {
			scope:republican.capital_county.culture = {
				save_scope_as = culture
			}
			scope:republican.capital_county.faith = {
				save_scope_as = faith
			}
			random = {
				chance = 75
				scope:republican.culture = {
					save_scope_as = culture
				}
				scope:republican.faith = {
					save_scope_as = faith
				}
			}
			create_character = {
				template = merchant_template
				culture = scope:culture
				faith = scope:faith
				employer = scope:republican
				gender_female_chance = location_faith_dominant_gender_female_chance
				age = {
					integer_range = {
						min = 30
						max = 60
					}
				}
				dynasty = generate
				save_scope_as = vassal
			}
			scope:vassal = {
				add_character_flag = { flag = gdp_becoming_patrician days = 1 }
			}
			create_dynamic_title = {
				name = NEW_VASSAL_PATRICIAN_TITLE_NAME
				tier = county
				type = title_type_merchant_family
				holder = scope:vassal
				article = DEFAULT_TITLE_NAME_ARTICLE
				save_scope_as = new_patrician_title
			}
			scope:new_patrician_title ?= {
				add_title_law = noble_family_succession_law
			}
			scope:vassal = {
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = vassal_change
				}
				change_liege = {
					liege = scope:republican
					change = scope:vassal_change
				}
				resolve_title_and_vassal_change = scope:vassal_change
				change_government = republic_government
				remove_character_flag = gdp_becoming_patrician
				domicile ?= {
					if = {
						limit = { is_domicile_type = merchant_company }
						gdp_patrician_domicile_level_2_effect = yes
					}
				}
				save_scope_as = joining_patrician # For message scope
				scope:republican ?= {
					send_interface_message = {
						type = msg_patrician_joined
						title = msg_patrician_joined_republic_title
						desc = msg_patrician_joined_republic_desc
						left_icon = scope:joining_patrician
					}
				}
				every_player = {
					limit = {
						liege = scope:republican
						this != scope:republican
					}
					send_interface_message = {
						type = msg_patrician_joined
						title = msg_patrician_joined_republic_title
						desc = msg_patrician_joined_republic_desc
						left_icon = scope:joining_patrician
					}
				}
				save_scope_value_as = {
					name = child_count
					value = {
						integer_range = {
							min = 2
							max = {
								value = 2
								if = {
									limit = { age > 35 }
									add = 1
								}
								if = {
									limit = { age > 40 }
									add = 1
								}
							}
						}
					}
				}
				if = {
					limit = {
						scope:child_count >= 1
					}
					while = {
						count = scope:child_count
						create_character = {
							template = merchant_template
							culture = scope:vassal.culture
							faith = scope:vassal.faith
							employer = scope:vassal
							age = {
								integer_range = {
									min = 10
									max = { value = scope:vassal.age subtract = 16 }
								}
							}
							gender_female_chance = 50
							dynasty = inherit
							dynasty_house = scope:vassal.house
							save_scope_as = child
						}
						scope:child = {
							add_to_list = children
						}
						switch = {
							trigger = is_female
							yes = {
								scope:child = {
									set_mother = scope:vassal
								}
							}
							no = {
								scope:child = {
									set_father = scope:vassal
								}
							}
						}
					}
				}
			}
		}
	}
}

gdp_trade_league_replace_missing_leader_effect = {
	if = {
		limit = {
			has_leading_title = no

				count >= 1
			}
		}
		save_scope_as = confederation
		scope:title.holder ?= { save_scope_as = former_league_leader }

		ordered_confederation_member = {
			limit = {
				is_alive = yes
				government_has_flag = government_is_republic
				OR = {
					primary_title ?= {
						title_confederation ?= scope:confederation
					}
					any_held_title = {
						title_confederation ?= scope:confederation
					}
				}
			}
			order_by = gdp_trade_league_automatic_leader_score_value
			max = 1
			check_range_bounds = no
			save_scope_as = new_league_leader
		}

		if = {
			limit = { exists = scope:new_league_leader }
			gdp_set_trade_league_leading_title_from_character_effect = yes
		}
	}
}

# Needs scope:confederation and scope:new_league_leader.
gdp_set_trade_league_leading_title_from_character_effect = {
	if = {
		limit = {
			scope:new_league_leader.primary_title ?= {
				title_confederation ?= scope:confederation
			}
		}
		scope:new_league_leader.primary_title = { save_scope_as = new_leading_title }
	}
	else = {
		scope:new_league_leader = {
			random_held_title = {
				limit = { title_confederation ?= scope:confederation }
				save_scope_as = new_leading_title
			}
		}
	}

	if = {
		limit = { exists = scope:new_leading_title }
		set_leading_title = scope:new_leading_title
		if = {
			limit = {
				NOT = { has_variable = is_hanseatic_league }
			}
			scope:new_league_leader = { save_scope_as = league_creator }
			set_coa = scope:new_leading_title
			set_confederation_name = TRADE_LEAGUE_BASE_NAME
		}

		every_player = {
			limit = { confederation ?= scope:confederation }
			if = {
				limit = { exists = scope:former_league_leader }
				send_interface_message = {
					type = msg_trade_league_leadership_changed
					title = msg_trade_league_leader_succeeded
					desc = msg_trade_league_leader_succeeded_desc
					left_icon = scope:new_league_leader
					right_icon = scope:former_league_leader
				}
			}
			else = {
				send_interface_message = {
					type = msg_trade_league_leadership_changed
					title = msg_trade_league_leader_succeeded
					desc = msg_trade_league_leader_succeeded_no_predecessor_desc
					left_icon = scope:new_league_leader
				}
			}
		}
	}
}

# Run on a patrician whose republic is gone
# Over the headroom only notable families become merchants, or every family if a player destroyed the republic
# The rest go to the pool
gdp_destroyed_republic_patrician_effect = {
	if = {
		limit = {
			OR = {
				gdp_merchant_companies_below_republic_headroom_trigger = yes
				$DESTROYER$ ?= { is_ai = no }
				gdp_merchant_company_keep_trigger = yes
			}
		}
		create_merchant_title_effect = {
			REASON = flag:republic_destroyed
			FLAVOR_CHAR = $DESTROYER$
		}
	}
	else = {
		save_temporary_scope_as = dissolved_patrician
		every_held_title = {
			limit = { is_noble_family_title = yes }
			scope:dissolved_patrician = { destroy_title = prev }
		}
		if = {
			limit = { is_pool_character = no }
			move_to_pool = yes
		}
	}
}

# Effect for converting to feudal from republic
convert_to_feudalism_from_republic_effect = {
	# Change government
	if = {
		limit = { ep3_is_clan_inclined_trigger = yes }
		change_government = clan_government

		every_vassal = {
			limit = {
				primary_title.tier >= tier_county
				is_landed = yes
				NOR = {
					government_has_flag = government_is_clan
					government_has_flag = government_is_theocracy
				}
			}
			hidden_effect = { change_government = clan_government }
		}
	}
	else = {
		change_government = feudal_government

		every_vassal = {
			limit = {
				primary_title.tier >= tier_county
				is_landed = yes
				NOR = {
					government_has_flag = government_is_feudal
					government_has_flag = government_is_theocracy
				}
			}
			hidden_effect = { change_government = feudal_government }
		}
	}
	
	# Turn patricians into merchants
	if = {
		limit = {
			any_vassal = {
				government_has_flag = government_is_republic
				is_landed = no
				NOT = { has_character_flag = government_republic_conversion_protection }
				any_held_title = { is_noble_family_title = yes }
			}
		}
		save_scope_as = destroyer
		custom_tooltip = {
			text = patricians_becomes_merchants_desc
			every_vassal = {
				limit = {
					government_has_flag = government_is_republic
					is_landed = no
					NOT = { has_character_flag = government_republic_conversion_protection }
					any_held_title = { is_noble_family_title = yes }
				}
				# Create Merchant
				set_variable = {
					name = republic_destroyer
					value = scope:destroyer
					weeks = 1
				}
				gdp_destroyed_republic_patrician_effect = { DESTROYER = scope:destroyer }
			}
		}
	}
}

gdp_find_deposed_republic_successor_effect = {
	scope:despot = {
		if = { # Next non-house in-line
			limit = {
				primary_title = {
					any_title_heir = {
						NOT = { house = scope:despot.house }
					}
				}
			}
			primary_title = {
				ordered_title_heir = {
					limit = { NOT = { house = scope:despot.house } }
					max = 1
					save_scope_as = republic_heir
				}
			}
		}
		else_if = { # Random other noble family
			limit = {
				any_noble_family = {
					holder = {
						house != scope:despot.house
						is_available_quick = {
							adult = yes
							imprisoned = no
							incapable = no
						}
					}
				}
			}
			random_noble_family = {
				limit = {
					holder = {
						house != scope:despot.house
						is_powerful_vassal = yes
						is_available_quick = {
							adult = yes
							imprisoned = no
							incapable = no
						}
					}
				}
				alternative_limit = {
					holder = {
						house != scope:despot.house
						is_available_quick = {
							adult = yes
							imprisoned = no
							incapable = no
						}
					}
				}
				weight = {
					base = 100
					modifier = {
						add = {
							value = "holder.opinion(scope:despot)"
							multiply = -1
						}
					}
					modifier = {
						holder = { has_relation_rival = scope:despot }
						add = 100
					}
					modifier = {
						holder = {
							OR = {
								is_close_family_of = scope:despot
								is_spouse_of = scope:despot
							}
						}
						add = -100
					}
					modifier = {
						exists = holder.joined_faction
						add = 100
					}
					house_relation_ai_accept_modifiers = {
						HOUSE_1_MEMBER = holder
			 			HOUSE_2_MEMBER = scope:despot
		 				MULTIPLIER = -1
					}
				}
				holder = { save_scope_as = republic_heir }
			}
		}
		else = { # Upstart
			create_character = {
				template = peasant_leader_stewardship_character
				dynasty = generate
				culture = scope:despot.capital_county.culture
				rite = scope:despot.capital_county.rite
				location = scope:despot.capital_province
				save_scope_as = republic_heir
			}
			scope:republic_heir = { add_character_flag = generated_character }
		}
	}
}
