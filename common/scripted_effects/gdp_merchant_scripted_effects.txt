# Overseer Martial Lifestyle Perk Effect
add_merchant_competitive_pressure_effect = {
	if = {
		limit = {
			government_has_flag = government_is_merchant
			has_perk = serve_the_crown_perk
		}
		custom_tooltip = $TOOLTIP$
		hidden_effect = {
			if = {
				limit = { NOT = { exists = var:merchant_competitive_pressure } }
				set_variable = {
					name = merchant_competitive_pressure
					value = $VALUE$
				}
			}
			else = {
				change_variable = {
					name = merchant_competitive_pressure
					add = $VALUE$
				}
			}
			if = {
				limit = {
					has_variable = merchant_competitive_pressure
				}
				save_scope_value_as = {
					name = merchant_competitive_pressure_overflow_value
					value = {
						value = var:merchant_competitive_pressure
						subtract = merchant_competitive_pressure_current_max_value
						min = 0
					}
				}
			}
			if = {
				limit = {
					OR = {
						NOT = { exists = var:merchant_competitive_pressure }
						var:merchant_competitive_pressure > merchant_competitive_pressure_current_max_value
					}
				}
				set_variable = {
					name = merchant_competitive_pressure
					value = merchant_competitive_pressure_current_max_value
				}
			}
		}
		if = {
			limit = {
				has_perk = absolute_control_perk
				exists = scope:merchant_competitive_pressure_overflow_value
				scope:merchant_competitive_pressure_overflow_value > 0
			}
			add_prestige = {
				value = scope:merchant_competitive_pressure_overflow_value
				multiply = merchant_dominant_position_prestige_per_pressure_value
			}
		}
	}
}

bailiff_collect_loan_payment_effect = {
	$LOAN$ = { save_scope_as = loan }
	scope:loan.story_owner = { save_scope_as = debtor }
	save_scope_value_as = {
		name = collected_gold
		value = {
			value = 0
			scope:liege ?= {
				add = bailiff_collect_loans_payment_value
			}
			max = scope:loan.var:amount_owed
			max = scope:debtor.gold
		}
	}
	if = {
		limit = { scope:loan.var:amount_owed <= scope:collected_gold }
		scope:loan = { repay_and_end_loan_story_effect = yes }
	}
	else = {
		scope:liege = {
			send_interface_toast = {
				type = event_royal_court_good_with_text
				title = bailiff_collect_loans_title
				desc = bailiff_collect_loans_notification_desc
				left_icon = root
				right_icon = scope:debtor
				scope:debtor = {
					pay_short_term_gold = {
						target = scope:liege
						gold = scope:collected_gold
					}
				}
				gdp_merchant_creditor_loan_acumen_effect = { GOLD = scope:collected_gold }
				hidden_effect = {
					scope:loan = {
						change_variable = {
							name = amount_owed
							add = {
								value = scope:collected_gold
								multiply = -1
							}
						}
						save_scope_value_as = {
							name = outstanding
							value = var:amount_owed
						}
					}
				}
				scope:loan = {
					custom_tooltip = bailiff_collect_loans_remaining_tt
				}
			}
		}
	}
}

		scope:title = {
			every_county_province = {
				every_province_domicile = {
					limit = {
						is_domicile_type = merchant_company
						owner ?= { government_has_flag = government_is_merchant } # Republics share the domicile type
					}
					owner = {
						save_scope_as = merchant
						scope:previous_holder ?= {
							if = { # stop confusing messages when a republic is destroyed
								limit = {
									NOT = { government_has_flag = government_is_merchant }
								}
								save_scope_as = old_suzerain
							}
						}
						if = {
							limit = { NOT = { exists = scope:old_suzerain } }
							suzerain ?= { save_scope_as = old_suzerain }
						}
						scope:new_holder ?= { save_scope_as = new_suzerain }
						if = {
							limit = { NOT = { exists = scope:new_suzerain } }
							domicile.domicile_location.county.holder ?= { save_scope_as = new_suzerain }
						}
						if = {
							limit = {
								exists = scope:new_suzerain
								scope:new_suzerain = { is_alive = yes }
								scope:merchant != scope:new_suzerain
								NOT = { suzerain ?= scope:new_suzerain }
							}
							start_tributary_interaction_effect = {
								TRIBUTARY = scope:merchant
								SUZERAIN = scope:new_suzerain
							}
						}
						if = {
							limit = {
								scope:old_suzerain ?= { is_alive = no }
								exists = scope:new_suzerain
								suzerain ?= scope:new_suzerain
							}
							gdp_merchant_charter_transfer_patron_effect = {
								OLD_PATRON = scope:old_suzerain
								NEW_PATRON = scope:new_suzerain
							}
						}
						trigger_event = {
							id = gdp_merchant.9011
							delayed = yes
						}
						clear_saved_scope = old_suzerain
					}
				}
			}
		}
	}
}

create_merchant_title_effect = {
	save_scope_as = new_merchant
	house ?= { save_scope_as = new_merchant_house }
	location ?= { save_scope_as = new_merchant_location }
	# Note our courtiers so that they don't escape.
	if = {
		limit = {
			OR = {
				government_has_flag = government_is_landless_adventurer
				is_landless_administrative = yes
			}
		}
		every_courtier = { add_to_list = courtiers_list }
	}
	# We save the reason for creating a Merchant House, for flavor purposes (memories and whatnot)
	save_scope_value_as = {
		name = merchant_creation_reason
		value = $REASON$
	}
	# We save the person who is taking over for us
	$FLAVOR_CHAR$ = { save_scope_as = merchant_flavor_char }
	custom_tooltip = become_merchant_tt
	if = {
		limit = {
			gdp_merchant_can_reclaim_house_company_title_trigger = no
			gdp_merchant_needs_cadet_for_new_company_trigger = yes
		}
		custom_tooltip = become_merchant_form_cadet_branch_desc
	}
	# ORDER OF OPERATION HERE IS IMPORTANT
	hidden_effect = {
		if = {
			limit = { exists = scope:merchant_flavor_char }
			set_variable = {
				name = pending_merchant_creation_reason
				value = scope:merchant_creation_reason
			}
			set_variable = {
				name = pending_merchant_flavor_char
				value = scope:merchant_flavor_char
			}
		}
		every_held_title = {
			limit = {
				is_noble_family_title = yes
				gdp_merchant_company_title_trigger = no
			}
			save_temporary_scope_as = old_noble_family_title
			scope:new_merchant = { destroy_title = scope:old_noble_family_title }
		}
		# Only create a new title if this character doesn't have one already (from this effect or from being a Patrician)
		if = {
			limit = {
				any_held_title = { gdp_merchant_company_title_trigger = yes }
			}
			random_held_title = {
				limit = { gdp_merchant_company_title_trigger = yes }
				save_scope_as = the_new_title
			}
		}
		# Also account for a company title they dropped before, say by becoming a landless adventurer
		# This effect saves the title to be reclaimed as the_new_title
		else_if = {
			limit = { gdp_merchant_can_reclaim_house_company_title_trigger = yes }
			gdp_merchant_reclaim_house_company_title_effect = yes
		}
		# Do we need to create a cadet branch? 
		else_if = {
			limit = { gdp_merchant_needs_cadet_for_new_company_trigger = yes }
			save_scope_as = new_house_head
			save_best_location_for_house_name_effect = yes
			create_cadet_branch = {
				name = cadet_name_style_county_dynasty
				save_scope_as = new_house
			}
			scope:new_house ?= { generate_coa = yes }
			# House titles require every potential successor to share the holder's house.
			every_close_family_member = {
				limit = {
					is_alive = yes
					NOT = { exists = house }
				}
				set_house = scope:new_house
			}
			# Create a new title
			create_dynamic_title = {
				name = merchant_name_001
				tier = county
				type = title_type_merchant_family
				holder = scope:new_merchant
				article = DEFAULT_TITLE_NAME_ARTICLE
				save_scope_as = the_new_title
			}
		}
		# Now, if all the above checks failed, and the_new_title doesn't exist...
		else = {
			# House titles require every potential successor to share the holder's house
			# Ruler-designer merchants often have generated lowborn parents or children with no house
			if = {
				limit = { exists = house }
				house = { save_scope_as = merchant_title_house }
				every_close_family_member = {
					limit = {
						is_alive = yes
						NOT = { exists = house }
					}
					set_house = scope:merchant_title_house
				}
			}
			# Create a new title
			create_dynamic_title = {
				name = merchant_name_001
				tier = county
				type = title_type_merchant_family
				holder = scope:new_merchant
				article = DEFAULT_TITLE_NAME_ARTICLE
				save_scope_as = the_new_title
			}
		}
		remove_variable = pending_merchant_creation_reason
		remove_variable = pending_merchant_flavor_char
		every_in_list = {
			list = courtiers_list
			limit = {
				is_alive = yes
				NOT = { liege_or_court_owner = scope:new_merchant }
			}
			scope:new_merchant = { add_courtier = prev }
		}
		# Destroy laamp title
		if = {
			limit = {
				any_held_title = { has_variable = adventurer_creation_reason }
			}
			destroy_laamp_effect = { ADVENTURER = scope:new_merchant }
		}
		# Transfer landed titles
		every_held_title = {
			title_tier > barony
			limit = {
				OR = {
					is_landless_type_title = no
					is_nomad_title = yes
				}
			}
			add_to_list = target_titles
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = no
		}
		if = {
			limit = {
				exists = scope:laamp_inheritor
			}
			every_in_list = {
				list = target_titles
				change_title_holder_include_vassals = {
					holder = scope:laamp_inheritor
					change = scope:change
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		# Finally, we change government type
		scope:new_merchant = {
			if = {
				limit = { is_ruler = yes }
				if = {
					limit = {
						NOT = { has_government = merchant_government }
					}
					change_government = merchant_government
				}
				if = {
					limit = {
						NOT = { has_realm_law = merchant_succession_law }
					}
					add_realm_law = merchant_succession_law
				}
				if = {
					limit = {
						has_realm_law = crown_authority_0
					}
					remove_realm_law = crown_authority_0
				}
				if = {
					limit = {
						has_realm_law = tribal_authority_0
					}
					remove_realm_law = tribal_authority_0
				}
			}
			if = {
				limit = {
					NOT = { has_trait = lifestyle_merchant }
				}
				add_trait = lifestyle_merchant
			}
			if = {
				limit = {
					is_ai = yes
					is_ruler = yes
				}
				# Set some starting Company purposes
				switch = {
					trigger = highest_skill
					learning = { add_realm_law_skip_effects = company_purpose_artisan }
					diplomacy = { add_realm_law_skip_effects = company_purpose_caravaneer }
					martial = { add_realm_law_skip_effects = company_purpose_creditor }
					intrigue = { add_realm_law_skip_effects = company_purpose_creditor }
				}
				set_starting_company_size_effect = yes

				# Populate conditional buildings, the domicile type already gives the standard ones
				# Merchants made at game start keep the level 1 buildings their domicile type gives them
				if = {
					limit = {
						is_from_ruler_designer = no
						NOT = { exists = scope:gdp_starting_merchant }
					}
					generate_appropriate_merchant_buildings = yes
				}
			}
			if = {
				limit = { is_ruler = yes }
				liege ?= {
					if = {
						limit = { this != scope:new_merchant }
						save_scope_as = old_liege
						if = {
							limit = { has_treasury = yes }
							enact_treasury_budgets_no_costs_no_opinion_effects = yes
						}
					}
				}
				if = {
					limit = { liege != scope:new_merchant }
					create_title_and_vassal_change = {
						type = independency
						save_scope_as = change
						add_claim_on_loss = no
					}
					becomes_independent = { change = scope:change }
					resolve_title_and_vassal_change = scope:change
				}
				start_tributary_interaction_effect = {
					TRIBUTARY = scope:new_merchant
					SUZERAIN = scope:new_merchant.domicile.domicile_location.county.holder
				}
			}
			if = {
				limit = { exists = scope:destroyer }
				trigger_event = gdp_merchant.9110
			}
		}
		# Then save some scopes on the new title for ease of reference
		scope:the_new_title ?= {
			if = {
				limit = { NOT = { has_variable = merchant_creation_reason } }
				set_variable = {
					name = merchant_creation_reason
					value = scope:merchant_creation_reason
				}
			}
			if = {
				limit = { NOT = { exists = var:merchant_flavor_char } }
				set_variable = {
					name = merchant_flavor_char
					value = scope:merchant_flavor_char
				}
			}
			# Set CoA
			set_coa = scope:new_merchant.house
			set_variable = {
				name = gdp_new_merchant_company
				years = 10
			}
		}
		# Slush fund. Ruler-designer merchants are set to 50 gold in on_game_start_after_lobby.
		if = {
			limit = { is_from_ruler_designer = no }
			add_gold = { value = 150 }
		}
		# Courtiers.
		trigger_event = gdp_merchant.1001
		# And tally our numbers for performance purposes.
		if = {
			limit = { is_ruler = yes }
			add_to_global_variable_list = {
				name = merchants_tally
				target = this.primary_title
			}
		}
		every_player = {
			limit = {
				NOT = {
					this = scope:new_merchant
					scope:new_merchant.var:enticed_merchant ?= this
				}
				NOT = { # we get it, every family here is now a merchant
					exists = scope:destroyer
					var:republic_destroyer ?= scope:destroyer
				}
				OR = {
					# Either they're important to us.
					is_close_or_extended_family_of = scope:new_merchant
					is_consort_of = scope:new_merchant
					has_important_relationship_with_character_trigger = { CHARACTER = scope:new_merchant }
					# Or they're close enough
					AND = {
						capital_province = { "squared_distance(scope:new_merchant.location)" <= squared_distance_medium }
						OR = {
							top_liege = scope:new_merchant.location.county.holder.top_liege
							any_neighboring_and_across_water_top_liege_realm_owner = { this = scope:new_merchant.location.county.holder.top_liege }
						}
					}
				}
			}
			if = {
				limit = {
					exists = scope:merchant_flavor_char
					scope:merchant_flavor_char != scope:new_merchant
				}
				send_interface_message = {
					type = msg_merchant_nearby
					title = nearby_merchant_created_message_title
					left_icon = scope:new_merchant
					right_icon = scope:merchant_flavor_char
					switch = {
						trigger = scope:merchant_creation_reason
						flag:republic_destroyed = { custom_tooltip = nearby_merchant_created_message_republic_destroyed_tt }
						flag:voluntary = { custom_tooltip = nearby_merchant_created_message_voluntary_tt }
						flag:fleeing_republic = { custom_tooltip = fleeing_republic_merchant_created_message_eradicate_tt }
						fallback = { custom_tooltip = nearby_merchant_created_message_fallback_tt }
					}
				}
			}
			else = {
				send_interface_message = {
					type = msg_merchant_nearby
					title = nearby_merchant_created_message_title
					left_icon = scope:new_merchant
					switch = {
						trigger = scope:merchant_creation_reason
						flag:republic_destroyed = { custom_tooltip = nearby_merchant_created_message_republic_destroyed_tt }
						flag:fleeing_republic = { custom_tooltip = fleeing_republic_merchant_created_message_eradicate_tt }
						fallback = { custom_tooltip = nearby_merchant_created_message_fallback_tt }
					}
				}
			}
		}
		scope:new_merchant = {
			every_councillor = {
				if = {
					limit = {
						NOR = {
							has_council_position = councillor_court_chaplain
							has_council_position = councillor_spouse
						}
					}
					scope:new_merchant = { fire_councillor = prev }
					remove_opinion = {
						modifier = fired_from_council_opinion
						target = scope:new_merchant
					}
				}
				else_if = {
					limit = {
						NOR = {
							is_close_or_extended_family_of = scope:new_merchant
							has_relation_lover = scope:new_merchant
							has_relation_friend = scope:new_merchant
							is_consort_of = scope:new_merchant
							has_council_position = councillor_spouse
							is_in_list = courtiers_list
						}
					}
					death = { death_reason = death_vanished }
				}
			}
			every_courtier = {
				limit = {
					NOR = {
						is_close_or_extended_family_of = scope:new_merchant
						has_relation_lover = scope:new_merchant
						has_relation_friend = scope:new_merchant
						is_consort_of = scope:new_merchant
						is_in_list = courtiers_list
					}
				}
				move_to_pool = yes
			}
			trigger_event = gdp_merchant.1001
		}
	}
}

merchant_trade_opportunity_save_lost_reason_effect = {
	if = {
		limit = {
			NOT = { exists = var:trade_opportunity_lost_reason }
		}
		if = {
			limit = {
				NOT = { exists = var:task_contract_route }
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:route_destroyed
			}
		}
		else_if = {
			limit = {
				OR = {
					# Employer death invalidates in code while the route can still name that same character.
					task_contract_employer = { is_alive = no }
					NOT = { var:task_contract_route.trader ?= task_contract_employer }
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:trader_changed
			}
		}
		else_if = {
			limit = {
				exists = var:task_contract_market
				var:task_contract_route = {
					NOT = {
						any_trade_market_on_route = {
							this = root.var:task_contract_market
						}
					}
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:market_left_route
			}
		}
		else_if = {
			limit = {
				OR = {
					has_task_contract_type = exploit_upheaval
					has_task_contract_type = alleviate_upheaval
				}
				var:task_contract_route = {
					NOT = {
						any_trade_market_on_route = {
							exploit_upheaval_market_modifier_trigger = yes
						}
					}
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:upheaval_ended
			}
		}
		else_if = {
			limit = {
				has_task_contract_type = support_guilds
				task_contract_taker = {
					NOT = { has_realm_law = company_purpose_artisan }
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:purpose_artisan
			}
		}
		else_if = {
			limit = {
				has_task_contract_type = seek_debtors
				task_contract_taker = {
					NOT = { has_realm_law = company_purpose_creditor }
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:purpose_creditor
			}
		}
		else_if = {
			limit = {
				OR = {
					has_task_contract_type = challenge_competitor
					has_task_contract_type = sabotage_competition
					has_task_contract_type = disrupt_charter
				}
				NOT = {
					task_contract_target.house.house_head ?= {
						gdp_is_merchant_or_unlanded_patrician_trigger = yes
						house != root.task_contract_taker.house
					}
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:competitor_gone
			}
		}
		else_if = {
			limit = {
				has_task_contract_type = disrupt_charter
				NOT = {
					task_contract_target.house.house_head ?= {
						merchant_has_charter_from_patron_trigger = {
							EQUIPPED = no
							INCLUDE_FORGERIES = yes
						}
					}
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:charter_gone
			}
		}
		else_if = {
			limit = {
				OR = {
					has_task_contract_type = entrepot_buy
					has_task_contract_type = entrepot_sell
				}
				NOT = {
					task_contract_location = { province_has_district = entrepot_location_district }
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:entrepot_gone
			}
		}
		else_if = {
			limit = {
				OR = {
					has_task_contract_type = negotiate_tolls
					has_task_contract_type = smuggle_goods
				}
				NOT = {
					task_contract_location.barony.trade_market = {
						pays_any_tolls_in_market_without_exception_trigger = { CHARACTER = root.task_contract_taker }
					}
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:tolls_ended
			}
		}
		else_if = {
			limit = {
				OR = {
					has_task_contract_type = silk_road_network
					has_task_contract_type = silk_road_caravan
				}
				OR = {
					NOT = {
						task_contract_taker = { merchant_is_silk_road_participant_trigger = yes }
					}
					var:task_contract_route = {
						NOT = {
							any_trade_market_on_route = {
								trade_market_owner.title_province = root.task_contract_location
								is_target_in_global_variable_list = {
									name = silk_road_markets
									target = this
								}
							}
						}
					}
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:silk_road
			}
		}
		else_if = {
			limit = {
				task_contract_taker = { is_imprisoned = yes }
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:merchant_imprisoned
			}
		}
		else_if = {
			limit = {
				task_contract_taker = { is_incapable = yes }
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:merchant_incapable
			}
		}
		else_if = {
			limit = {
				task_contract_taker = {
					NOT = { government_has_flag = government_is_merchant }
				}
			}
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:merchant_government
			}
		}
	}
	if = {
		limit = {
			NOT = { var:trade_opportunity_lost_reason ?= flag:route_destroyed }
			exists = var:task_contract_route
		}
		var:task_contract_route = { save_scope_as = trade_opportunity_route }
	}
}

merchant_trade_opportunity_send_lost_toast_effect = {
	if = {
		limit = { exists = scope:task_contract.var:trade_opportunity_lost_reason }
		switch = {
			trigger = scope:task_contract.var:trade_opportunity_lost_reason
			flag:route_destroyed = {

destroy_merchant_effect = {
	$MERCHANT$ = {
		save_scope_as = the_merchant
		hidden_effect = {
			every_courtier = {
				gdp_merchant_courtier_becomes_associate_effect = yes
			}
		}
		# Liquidate Company
		custom_tooltip = {
			text = destroy_merchant_effect.tt.domicile_liquidated
			gdp_merchant_liquidate_company_effect = yes
		}
		# Destroy your main title.
		every_held_title = {
			limit = { has_variable = merchant_creation_reason }
			save_scope_as = merchant_title
			scope:the_merchant = { destroy_title = scope:merchant_title }
		}
		# The reset picks the best government for them, but administrative rulers should stay administrative, so we skip it for them
		if = {
			limit = {
				NOR = {
					government_has_flag = government_is_administrative
					government_has_flag = government_is_nomadic
					government_has_flag = government_is_steppe_admin
					government_has_flag = government_is_wanua
					government_has_flag = government_is_mandala
					government_has_flag = government_is_celestial
					government_has_flag = government_is_meritocratic
					government_has_flag = government_is_japan_feudal
				}
			}
			admin_government_reset_effect = yes #this only contains Clan, Tribal, Nomadic and Feudal
		}
		# Now we reforge any domicile artifacts
		# Artifact feature triggers read scope:owner, so it must be the merchant during reforging
		scope:owner ?= { save_scope_as = destroy_merchant_previous_owner }
		save_scope_as = owner
		if = {
			limit = {
				any_character_artifact = { artifact_type = library_item }
			}
			every_character_artifact = {
				limit = { artifact_type = library_item }
				reforge_artifact = { type = book }
			}
		}
		if = {
			limit = {
				any_character_artifact = { artifact_type = curio }
			}
			every_character_artifact = {
				limit = { artifact_type = curio }
				reforge_artifact = { type = pedestal }
			}
		}
		if = {
			limit = {
				any_character_artifact = { artifact_type = creature }
			}
			every_character_artifact = {
				limit = { artifact_type = creature }
				set_variable = {
					name = og_owner
					value = artifact_owner
				}
				# Animal skull features read the animal type from scope:owner, as set by hunts
				scope:the_merchant = {
					set_variable = {
						name = animal_type
						value = flag:none
						days = 1
					}
				}
				if = {
					limit = { has_variable = animal_type }
					scope:the_merchant = {
						set_variable = {
							name = animal_type
							value = prev.var:animal_type
							days = 1
						}
					}
				}
				reforge_creature_artifact_effect = yes
			}
			remove_variable = animal_type
		}
		scope:destroy_merchant_previous_owner ?= { save_scope_as = owner }
	}
}

# Turns a landless merchant into a patrician family of $LIEGE$'s republic, keeping the company.
gdp_merchant_becomes_patrician_effect = {
	save_scope_as = new_patrician_merchant
	save_scope_as = the_merchant
	$LIEGE$ = { save_scope_as = new_patrician_liege }

	# Court positions fall away with the government change, so stamp the Associate trait first.
	hidden_effect = {
		every_courtier = {
			gdp_merchant_courtier_becomes_associate_effect = yes
		}
	}

	# tributary_merchant cannot be broken from the merchant's side, so end it before swearing fealty.
	if = {
		limit = { is_tributary = yes }
		hidden_effect = { end_tributary = yes }
	}

	every_character_artifact = {
		limit = { has_variable = charter_patron }
		save_temporary_scope_as = surrendered_charter
		scope:new_patrician_merchant = { destroy_artifact = scope:surrendered_charter }
	}

	gdp_merchant_forgive_debtors_effect = yes

	if = {
		limit = {
			NOT = { liege ?= scope:new_patrician_liege }
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = new_patrician_vassal_change
		}
		change_liege = {
			liege = scope:new_patrician_liege
			change = scope:new_patrician_vassal_change
		}
		resolve_title_and_vassal_change = scope:new_patrician_vassal_change
	}

	change_government = republic_government
	# GDP music cue, only plays for the local player
	play_music_cue = mx_cue_the_rise_of_a_nation

	hidden_effect = {
		domicile ?= {
			if = {
				limit = {
					NOT = { domicile_location = scope:new_patrician_liege.capital_province }
				}
				move_domicile = scope:new_patrician_liege.capital_province
			}
		}
	}

	gdp_republic_restore_family_succession_law_effect = yes

	every_held_title = {
		limit = { gdp_merchant_company_title_trigger = yes }
		set_title_name_dynamic = NEW_PATRICIAN_TITLE_NAME_FROM_COMPANY
		# The title survives, so it never hits the on_title_destroyed prune that maintains merchants_tally.
		save_temporary_scope_as = departing_company_title
		if = {
			limit = { has_global_variable_list = merchants_tally }
			remove_list_global_variable = {
				name = merchants_tally
				target = scope:departing_company_title
			}
		}
	}
}

# Settles a landless merchant as an unlanded noble family of $GIVER$'s administrative realm
# Company and estate are different domicile types, so we destroy the company title
gdp_merchant_procures_estate_effect = {
	save_scope_as = settling_merchant
	$GIVER$ = { save_scope_as = estate_giver }

	# tributary_merchant cannot be broken from the merchant's side, so end it before swearing fealty.
	if = {
		limit = { is_tributary = yes }
		hidden_effect = { end_tributary = yes }
	}

	hidden_effect = {
		every_character_artifact = {
			limit = { has_variable = charter_patron }
			save_temporary_scope_as = surrendered_charter
			scope:settling_merchant = { destroy_artifact = scope:surrendered_charter }
		}
		gdp_merchant_forgive_debtors_effect = yes
		# The government arrives via give_noble_family_title, so on_government_change cannot be relied on.
		gdp_invalidate_merchant_task_contracts_on_government_change_effect = yes
		every_courtier = { add_to_list = courtiers_list }
	}

	destroy_merchant_effect = { MERCHANT = scope:settling_merchant }

	# Checked only after the company is gone, since the company itself is a noble family title.
	if = {
		limit = {
			house ?= {
				any_house_member = {
					this != scope:settling_merchant
					top_liege = scope:estate_giver.top_liege
					any_held_title = { is_noble_family_title = yes }
				}
			}
		}
		found_cadet_house_decision_effect = {
			CHARACTER = scope:settling_merchant
			PRESTIGE = major_prestige_gain
		}
	}

	hidden_effect = {
		scope:estate_giver.top_liege = {
			add_courtier = scope:settling_merchant
			scope:settling_merchant = { return_to_court = yes }
		}
	}

	create_noble_family_effect = { GOVERNMENT_GIVER = scope:estate_giver }

	hidden_effect = {
		every_in_list = {
			list = courtiers_list
			limit = {
				NOT = { liege_or_court_owner = scope:settling_merchant }
			}
			scope:settling_merchant = { add_courtier = prev }
		}
		# And boot any new autogenned arrivals.
		every_courtier = {
			limit = {
				is_lowborn = yes
				NOR = {
					is_in_list = courtiers_list
					is_close_or_extended_family_of = scope:settling_merchant
					is_consort_of = scope:settling_merchant
					has_important_relationship_trigger = { CHARACTER = scope:settling_merchant }
				}
			}
			silent_disappearance_effect = yes
		}
	}

	# create_noble_family_effect already swears fealty when it resolves a liege of its own.
	if = {
		limit = { top_liege = this }
		create_title_and_vassal_change = {
			type = swear_fealty
			add_claim_on_loss = no
			save_scope_as = estate_vassal_change
		}
		change_liege = {
			liege = scope:estate_giver.top_liege
			change = scope:estate_vassal_change
		}
		resolve_title_and_vassal_change = scope:estate_vassal_change
	}

	hidden_effect = {
		domicile ?= {
			if = {
				limit = {
					NOT = { domicile_location = scope:estate_giver.capital_province }
				}
				move_domicile = scope:estate_giver.capital_province
			}
		}
	}

	add_character_modifier = {
		modifier = ep3_administrative_outsider_modifier
		years = 15
	}
}

new_republic_population_effect = {
	save_scope_as = republican
	set_variable = {
		name = new_republic_invited_count
		value = 0
	}

	# Step 1: Find eligible AI merchants domiciled in the new capital's duchy
	# Skip those with negative house relations (feud or rivalry) or at war with the founder
	hidden_effect = {
		scope:republican.capital_county.duchy ?= {
			every_de_jure_county = {
				every_county_province = {
					every_province_domicile = {
						limit = {
							owner ?= {
								gdp_merchant_ai_logic_valid_trigger = yes
								this != scope:republican
								NOT = {
									is_at_war_with = scope:republican
								}
								NOT = {
									house ?= {
										any_house_relation = {
											any_relation_house = {
												scope:republican.house ?= this
											}
											OR = {
												has_house_relation_level = rivalry
												has_house_relation_level = feud
											}
										}
									}
								}
								# Founder's faith is not considered worse than astray
								faith = {
									faith_hostility_level = {
										target = scope:republican.faith
										value <= faith_astray_level
									}
								}
								# Culture is tolerated
								OR = {
									culture = scope:republican.culture
									culture = {
										OR = {
											has_same_culture_heritage = scope:republican.culture
											has_same_culture_language = scope:republican.culture
										}
										"cultural_acceptance(scope:republican.culture)" >= 25
									}
									AND = {
										knows_language_of_character = scope:republican
										"culture.cultural_acceptance(scope:republican.culture)" >= 50
									}
									"culture.cultural_acceptance(scope:republican.culture)" >= 75
								}
							}
						}
						owner ?= {
							add_to_list = invited_merchants
						}
					}
				}
			}
		}
	}

	# Invite existing merchants — they already have merchant titles, only need liege+government
	if = {
		limit = {
			any_in_list = {
				list = invited_merchants
			}
		}
		every_in_list = {
			list = invited_merchants
			save_scope_as = vassal
			hidden_effect = {
				gdp_merchant_becomes_patrician_effect = { LIEGE = scope:republican }
				scope:republican = {
					change_variable = {
						name = new_republic_invited_count
						add = 1
					}
				}
				scope:vassal = { save_scope_as = joining_patrician } # For message scope
				scope:republican ?= {
					send_interface_message = {
						type = msg_patrician_joined
						title = msg_patrician_joined_republic_title
						desc = msg_patrician_joined_republic_desc
						left_icon = scope:joining_patrician
					}
				}
			}
			if = {
				limit = { exists = scope:portrait_2 }
				save_scope_as = portrait_3
			}
			else_if = {
				limit = { exists = scope:portrait_1 }
				save_scope_as = portrait_2
			}
			else = {
				save_scope_as = portrait_1
			}
		}
	}

	# Step 2: If fewer than 2 merchants were invited, generate the remainder dynamically
	if = {
		limit = {
			exists = var:new_republic_invited_count # tooltip error suppression when invoked via decision
			var:new_republic_invited_count < 2
		}
		save_scope_value_as = {
			name = generate_count
			value = {
				value = 2
				subtract = var:new_republic_invited_count
			}
		}
		hidden_effect = {
			while = {
				count = scope:generate_count
				scope:republican.capital_county.culture = {
					save_scope_as = culture
				}
				scope:republican.capital_county.rite = {
					save_scope_as = rite
				}
				random = {
					chance = 75
					scope:republican.culture = {
						save_scope_as = culture
					}
					scope:republican.rite = {
						save_scope_as = rite
					}
				}
				create_character = {
					template = generated_patrician_template
					culture = scope:culture
					rite = scope:rite
					employer = scope:republican
					gender_female_chance = location_faith_dominant_gender_female_chance
					age = {
						integer_range = {
							min = 30
							max = 60
						}
					}
					dynasty = generate
					save_scope_as = parent
				}
				scope:parent ?= {
					add_to_list = new_arrivals
				}
			}
		}
		every_in_list = {
			list = new_arrivals
			save_scope_as = vassal
			hidden_effect = {
				add_character_flag = { flag = gdp_becoming_patrician days = 1 }
				create_dynamic_title = {
					name = NEW_VASSAL_PATRICIAN_TITLE_NAME
					tier = county
					type = title_type_merchant_family
					holder = scope:vassal
					article = DEFAULT_TITLE_NAME_ARTICLE
					save_scope_as = new_patrician_title
				}
				scope:new_patrician_title ?= {
					add_title_law = noble_family_succession_law
				}
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = vassal_change
				}
				change_liege = {
					liege = scope:republican
					change = scope:vassal_change
				}
				resolve_title_and_vassal_change = scope:vassal_change
				change_government = republic_government
				remove_character_flag = gdp_becoming_patrician
				domicile ?= {
					if = {
						limit = { is_domicile_type = merchant_company }
						gdp_patrician_domicile_level_2_effect = yes
					}
				}
				scope:vassal = { save_scope_as = joining_patrician } # For message scope
				scope:republican ?= {
					send_interface_message = {
						type = msg_patrician_joined
						title = msg_patrician_joined_republic_title
						desc = msg_patrician_joined_republic_desc
						left_icon = scope:joining_patrician
					}
				}
				save_scope_value_as = {
					name = child_count
					value = {
						integer_range = {
							min = 2
							max = 4
						}
					}
				}
				save_scope_value_as = {
					name = min_age
					value = 10
				}
				save_scope_value_as = {
					name = max_age
					value = {
						value = scope:vassal.age
						add = -16
					}
				}
				if = {
					limit = {
						scope:max_age >= scope:min_age
					}
					while = {
						count = scope:child_count
						create_character = {
							template = generated_patrician_template
							culture = scope:vassal.culture
							rite = scope:vassal.rite
							employer = scope:vassal
							age = {
								integer_range = {
									min = scope:min_age
									max = scope:max_age
								}
							}
							gender_female_chance = 50
							dynasty = inherit
							dynasty_house = scope:vassal.house
							save_scope_as = child
						}
						scope:child = {
							add_to_list = children
						}
						switch = {
							trigger = is_female
							yes = {
								scope:child = {
									set_mother = scope:vassal
								}
							}
							no = {
								scope:child = {
									set_father = scope:vassal
								}
							}
						}
					}
				}
			}
			if = {
				limit = {
					exists = scope:portrait_2
				}
				save_scope_as = portrait_3
			}
			else_if = {
				limit = {
					exists = scope:portrait_1
				}
				save_scope_as = portrait_2
			}
			else = {
				save_scope_as = portrait_1
			}
		}
	}
}

# The route's first traded good, when its category has an xp track. Otherwise fall back to a purpose category.
set_starting_trade_expertise_effect = {
	save_temporary_scope_as = company
	holder ?= {
		if = {
			limit = {
				is_ai = yes
				is_alive = yes
			}
			random_trade_route = {
				# every_traded_good is ordered, so we take the primary good first
				every_traded_good = {
					limit = {
						scope:company = { NOT = { has_variable = checked_first_traded_good } }
					}
					scope:company = { set_variable = { name = checked_first_traded_good value = yes } }
					trade_good_need_category = {
						if = {
							limit = { has_trade_good_category_xp_track = yes }
							save_temporary_scope_as = starting_expertise_category
							scope:company = { set_variable = { name = apply_starting_expertise value = yes } }
						}
					}
				}
			}
			if = {
				limit = {
					scope:company = { NOT = { has_variable = apply_starting_expertise } }
				}
				# Artisan and Trader: luxury items. Caravaneer: luxury consumables. Creditor: precious metals.
				if = {
					limit = { has_realm_law = company_purpose_caravaneer }
					trade_good_category:trade_category_luxury_consumption = {
						save_temporary_scope_as = starting_expertise_category
					}
				}
				else_if = {
					limit = { has_realm_law = company_purpose_creditor }
					trade_good_category:trade_category_precious_metals = {
						save_temporary_scope_as = starting_expertise_category
					}
				}
				else = {
					trade_good_category:trade_category_luxury_items = {
						save_temporary_scope_as = starting_expertise_category
					}
				}
				scope:company = { set_variable = { name = apply_starting_expertise value = yes } }
			}
		}
	}
	remove_variable = checked_first_traded_good
	if = {
		limit = { has_variable = apply_starting_expertise }
		remove_variable = apply_starting_expertise
		if = {
			limit = { game_start_date < 1066.1.1 }
			add_trade_good_category_xp = {
				target = scope:starting_expertise_category
				value = early_starting_expertise_value
			}
		}
		else_if = {
			limit = { game_start_date < 1178.10.1 }
			add_trade_good_category_xp = {
				target = scope:starting_expertise_category
				value = mid_starting_expertise_value
			}
		}
		else = {
			add_trade_good_category_xp = {
				target = scope:starting_expertise_category
				value = late_starting_expertise_value
			}
		}
	}
}

# Run in the scope of a merchant whose company is being absorbed. Recreates each of their trade monopolies under $NEW_OWNER$.
# The old monopoly has to be queued for destruction first, since a market only allows one monopoly per category.
gdp_merchant_recreate_trade_monopolies_for_new_owner_effect = {
	if = {
		limit = { character_has_trade_monopoly = yes }
		custom_description = {
			text = merge_company_subsumed_trade_monopolies_tt
			subject = this
			object = $NEW_OWNER$
			$NEW_OWNER$ = {
				every_trade_monopoly_for_character = {
					add_to_temporary_list = gdp_merchant_new_owner_old_monopolies
				}
			}
			every_trade_monopoly_for_character = {
				add_to_temporary_list = gdp_merchant_monopolies_to_recreate
			}
			every_in_list = {
				list = gdp_merchant_monopolies_to_recreate
				monopoly_trade_market = { save_temporary_scope_as = recreated_monopoly_market }
				destroy_trade_monopoly = yes
				if = {
					limit = { is_complete_monopoly = yes }
					if = {
						limit = { monopoly_type = exploitative }
						create_trade_monopoly = {
							type = exploitative
							is_complete = yes
							owning_character = $NEW_OWNER$
							trade_market = scope:recreated_monopoly_market
						}
					}
					else = {
						create_trade_monopoly = {
							type = normal
							is_complete = yes
							owning_character = $NEW_OWNER$
							trade_market = scope:recreated_monopoly_market
						}
					}
				}
				else = {
					monopoly_trade_good_category = { save_temporary_scope_as = recreated_monopoly_category }
					if = {
						limit = { monopoly_type = exploitative }
						create_trade_monopoly = {
							type = exploitative
							owning_character = $NEW_OWNER$
							trade_market = scope:recreated_monopoly_market
							trade_good_category = scope:recreated_monopoly_category
						}
					}
					else = {
						create_trade_monopoly = {
							type = normal
							owning_character = $NEW_OWNER$
							trade_market = scope:recreated_monopoly_market
							trade_good_category = scope:recreated_monopoly_category
						}
					}
				}
			}
			$NEW_OWNER$ = {
				every_trade_monopoly_for_character = {
					limit = {
						is_complete_monopoly = no
						NOT = { is_in_list = gdp_merchant_new_owner_old_monopolies }
					}
					save_scope_as = new_monopoly_scope
					set_trade_monopoly_name = new_good_character_monopoly_name
				}
			}
		}
	}
}

				}

						modifier = hostile_takeover_opinion

					}

				}
			}
		}

			send_interface_toast = {
				type = event_toast_effect_neutral
				title = hostile_takeover_client_title
				left_icon = scope:takeover
				right_icon = scope:subsumed

			}

		pay_short_term_gold = {
			gold = scope:subsumed.gold
			target = scope:takeover
		}
	}
	scope:takeover = {
		if = {
			limit = { has_hook = scope:subsumed }
			use_hook = scope:subsumed
		}
		if = {
			limit = { has_hook = scope:subsumed_patron }
			use_hook = scope:subsumed_patron
		}
		house = {
			add_unity_value = {
				value = -50
				character = scope:takeover
				desc = merchant_unity_takeover_loss
			}
		}
	}
	scope:subsumed = {
		if = {
			limit = { has_relation_best_friend = scope:takeover }
			remove_relation_best_friend = scope:takeover
		}
		else_if = {
			limit = { has_relation_friend = scope:takeover }
			remove_relation_friend = scope:takeover
		}
		if = {
			limit = {
				can_set_relation_nemesis_trigger = { CHARACTER = scope:takeover }
			}
			set_relation_nemesis = { reason = hostile_takeover target = scope:takeover }
		}
		else_if = {
			limit = {
				can_set_relation_rival_trigger = { CHARACTER = scope:takeover }
			}
			set_relation_rival_effect = {
			    TARGET = scope:takeover
			    REASON = hostile_takeover
			}
		}
		add_opinion = {
			modifier = hostile_takeover_opinion
			target = scope:takeover
		}
		house ?= {
			if = {
				limit = {
					NOT = { has_house_relation_with = scope:takeover.house }
				}
				set_house_relation = {
					target = scope:takeover.house
					level = feud
					description = house_relation_reason_hostile_takeover_desc
				}
			}
			else = {
				change_house_relation_effect = {
					HOUSE = scope:takeover.house
					VALUE = house_relation_damage_major_value
					REASON = hostile_takeover
					CHAR = scope:takeover
					TARGET_CHAR = scope:subsumed
					TITLE = scope:dummy_gender
				}
			}
		}
	}
}

	}

		limit = { NOT = { exists = scope:tgf_market } }
		scope:tgf_route = {
			random_trade_market_on_route = {
				save_scope_as = tgf_market
			}

# Rolls a trade-good flavor artifact's rarity into scope:quality and scope:wealth, and its tier (1-4) into scope:tgf_artifact_tier
# $RARITY$ = random, masterwork or illustrious
gdp_tgf_roll_artifact_rarity_effect = {
	save_scope_value_as = {
		name = tgf_artifact_rarity
		value = flag:$RARITY$
	}
	if = {
		limit = { scope:tgf_artifact_rarity = flag:illustrious }
		set_artifact_rarity_illustrious = yes
		save_scope_value_as = { name = tgf_artifact_tier value = 4 }
	}
	else_if = {
		limit = { scope:tgf_artifact_rarity = flag:masterwork }
		set_artifact_rarity_masterwork = yes
		save_scope_value_as = { name = tgf_artifact_tier value = 2 }
	}
	else = {
		hidden_effect = {
			random_list = {
				50 = {
					set_artifact_rarity_common = yes
					save_scope_value_as = { name = tgf_artifact_tier value = 1 }
				}
				35 = {
					set_artifact_rarity_masterwork = yes
					save_scope_value_as = { name = tgf_artifact_tier value = 2 }
				}
				15 = {
					set_artifact_rarity_famed = yes
					save_scope_value_as = { name = tgf_artifact_tier value = 3 }
				}
			}
		}
	}
}

# Adds one random trade modifier of tier $TIER$ that the artifact does not have yet. this = artifact
gdp_tgf_add_one_artifact_trade_modifier_effect = {
	random_list = {
		10 = {
			trigger = { NOT = { has_variable = tgf_mod_expertise } }
			add_artifact_modifier = gdp_tgf_artifact_expertise_$TIER$_modifier
			set_variable = tgf_mod_expertise
		}
		10 = {
			trigger = { NOT = { has_variable = tgf_mod_competitiveness } }
			add_artifact_modifier = gdp_tgf_artifact_competitiveness_$TIER$_modifier
			set_variable = tgf_mod_competitiveness
		}
		10 = {
			trigger = { NOT = { has_variable = tgf_mod_capacity } }
			add_artifact_modifier = gdp_tgf_artifact_capacity_$TIER$_modifier
			set_variable = tgf_mod_capacity
		}
		10 = {
			trigger = { NOT = { has_variable = tgf_mod_trade_income } }
			add_artifact_modifier = gdp_tgf_artifact_trade_income_$TIER$_modifier
			set_variable = tgf_mod_trade_income
		}
		10 = {
			trigger = { NOT = { has_variable = tgf_mod_route_length } }
			add_artifact_modifier = gdp_tgf_artifact_route_length_$TIER$_modifier
			set_variable = tgf_mod_route_length
		}
	}
}

# Picks a route and a non-livestock good on it, then feeds the widget
# Saves scope:tgf_route, scope:tgf_good, scope:tgf_category
gdp_tgf_setup_packable_good_effect = {
	random_trade_route = {
		limit = {
			is_commission_route = no
			any_traded_good = { gdp_tgf_not_livestock_good_trigger = yes }
		}
		save_scope_as = tgf_route
		random_traded_good = {
			limit = { gdp_tgf_not_livestock_good_trigger = yes }
			save_scope_as = tgf_good
		}
	}
	scope:tgf_good = {
		trade_good_need_category = { save_scope_as = tgf_category }
	}
	gdp_tgf_set_widget_good_effect = { GOOD = scope:tgf_good }
}

# Saves bg_override_char as $CHAR$ when they stand in $PROVINCE$
# Otherwise any pool character there, so place backgrounds show that province
gdp_tgf_set_background_location_effect = {
	$PROVINCE$ ?= { save_scope_as = tgf_bg_province }
	if = {
		limit = {
			exists = scope:tgf_bg_province
			$CHAR$ ?= { location ?= scope:tgf_bg_province }
		}
		$CHAR$ = { save_scope_as = bg_override_char }
	}
	else_if = {
		limit = {
			exists = scope:tgf_bg_province
			any_pool_character = { province = scope:tgf_bg_province }
		}
		random_pool_character = {
			province = scope:tgf_bg_province
			save_scope_as = bg_override_char
		}
	}
}

# Force a Merchant Company out of the Expeller's realm and under a new Patron
gdp_evict_merchant_from_realm_effect = {
	$MERCHANT$ = { save_temporary_scope_as = merchant }
	$EXPELLER$ = { save_scope_as = expeller }
	scope:merchant.suzerain ?= { save_temporary_scope_as = old_patron }
	if = {
		limit = { NOT = { exists = scope:old_patron } }
		scope:expeller = { save_temporary_scope_as = old_patron }
	}
	scope:expeller.top_liege = { save_temporary_scope_as = expelled_realm }
	scope:merchant.domicile.domicile_location = { save_temporary_scope_as = origin_province }

	# Prefer a neighboring realm county with a trade market
	scope:expeller = {
		random_neighboring_and_across_water_top_liege_realm_owner = {
			limit = {
				NOT = { this = scope:expelled_realm }
				is_landed = yes
				any_realm_county = {
					exists = title_province.barony.trade_market
					NOT = { holder.top_liege = scope:expelled_realm }
				}
			}
			save_temporary_scope_as = destination_realm_owner
			random_realm_county = {
				limit = {
					exists = title_province.barony.trade_market
					NOT = { holder.top_liege = scope:expelled_realm }
				}
				weight = {
					base = 1
					modifier = { add = prosperity_level }
				}
				holder = { save_temporary_scope_as = new_patron }
				title_province = { save_temporary_scope_as = eviction_destination }
			}
		}
	}

	# Fallback: neighboring county title with a trade market outside the realm
	if = {
		limit = {
			NOR = {
				exists = scope:eviction_destination
				exists = scope:new_patron
			}
		}
		scope:expeller = {
			random_character_to_title_neighboring_and_across_water_county = {
				limit = {
					exists = title_province.barony.trade_market
					exists = holder
					NOT = { holder.top_liege = scope:expelled_realm }
				}
				holder = { save_temporary_scope_as = new_patron }
				title_province = { save_temporary_scope_as = eviction_destination }
			}
		}
	}

	# Last resort: nearest empire de jure county with a trade market outside the realm
	if = {
		limit = {
			NOR = {
				exists = scope:eviction_destination
				exists = scope:new_patron
			}
			exists = scope:origin_province
		}
		ordered_empire = {
			limit = {
				any_de_jure_county = {
					exists = title_province.barony.trade_market
					exists = holder
					NOT = { holder.top_liege = scope:expelled_realm }
				}
			}
			order_by = {
				title_capital_county = { add = "squared_distance(scope:origin_province)" }
				multiply = -1
			}
			random_de_jure_county = {
				limit = {
					exists = title_province.barony.trade_market
					exists = holder
					NOT = { holder.top_liege = scope:expelled_realm }
				}
				holder = { save_temporary_scope_as = new_patron }
				title_province = { save_temporary_scope_as = eviction_destination }
			}
		}
	}

	if = {
		limit = {
			exists = scope:merchant
			exists = scope:expeller
			exists = scope:new_patron
			exists = scope:eviction_destination
			exists = scope:eviction_destination.barony.trade_market
		}
		# Destroy any charter granted by the expeller
		if = {
			limit = {
				scope:old_patron ?= scope:expeller
				scope:merchant = {
					merchant_has_charter_from_patron_trigger = {
						EQUIPPED = no
						INCLUDE_FORGERIES = yes
					}
				}
			}
			merchant_charter_destroy_held_charters_effect = {
				MERCHANT = scope:merchant
				PATRON = scope:expeller
			}
		}
		scope:merchant = {
			add_opinion = {
				target = scope:expeller
				modifier = demanded_eviction_opinion
			}
			# Skip commission break fallout when routes are destroyed by eviction
			add_character_flag = {
				flag = being_evicted
				months = 1 # Guardrail if the travel route invalidates
			}
			# AI grace period, a crime does not go away with the move
			add_character_flag = {
				flag = gdp_recently_evicted
				years = 10
			}
			set_variable = {
				name = evicting_ruler
				value = scope:expeller
			}
			set_variable = {
				name = new_patron
				value = scope:new_patron
			}
			set_variable = {
				name = new_capital
				value = scope:eviction_destination
			}
			hidden_effect = {
				if = {
					limit = { is_ai = no }
					trigger_event = {
						id = gdp_merchant.9005
						delayed = no
					}
				}
			}
			current_travel_plan ?= {
				if = {
					limit = { travel_plan_owner ?= scope:merchant }
					if = {
						limit = { can_cancel = yes }
						cancel_travel_plan = yes
					}
				}
				else = { remove_character = scope:merchant }
			}
			start_travel_plan = {
				destination = scope:eviction_destination
				on_start_event = gdp_merchant.9000
				on_start_on_action = on_travel_relocation_start
				on_arrival_on_action = on_travel_relocation_end
				on_arrival_destinations = last
				return_trip = no
				travel_with_domicile = no
				can_cancel_planning = no
				players_use_planner = no
			}
		}
		hidden_effect = {
			scope:expeller = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = evict_merchant_interaction_toast.t
					left_icon = scope:merchant
					right_icon = scope:eviction_destination.barony
					custom_tooltip = evict_merchant_interaction_toast.desc
				}
			}
		}
	}
}

# Trade route scope, adds one maturity stack for every milestone reached since the last update
# A route with no stacks variable predates the system, so we start counting from its current milestone
# Improved routes go into gdp_maturity_gained_routes, and the caller sends the message
gdp_trade_route_maturity_update_effect = {
	if = {
		limit = { NOT = { has_variable = gdp_trade_route_maturity_stacks } }
		set_variable = {
			name = gdp_trade_route_maturity_stacks
			value = gdp_trade_route_maturity_milestones_value
		}
	}
	else = {
		save_temporary_scope_value_as = {
			name = maturity_new_stacks
			value = {
				value = gdp_trade_route_maturity_milestones_value
				subtract = var:gdp_trade_route_maturity_stacks
			}
		}
		if = {
			limit = { scope:maturity_new_stacks > 0 }
			while = {
				count = scope:maturity_new_stacks
				add_trade_route_modifier = gdp_trade_route_maturity_modifier
			}
			change_variable = {
				name = gdp_trade_route_maturity_stacks
				add = scope:maturity_new_stacks
			}
			add_to_temporary_list = gdp_maturity_gained_routes
		}
	}
}

# Historical company expertise: level 5 plus one per decade since founding, capped at level 50
gdp_historical_company_expertise_effect = {
	title:$TITLE$ = {
		if = {
			limit = { exists = holder }
			add_trade_good_category_xp = {
				target = trade_good_category:trade_category_$CATEGORY$
				value = {
					value = current_year
					subtract = $FOUNDED$
					divide = 10
					add = 5
					max = 50
					multiply = define:NTrade|TRADE_CATEGORY_XP_LEVEL_THRESHOLD
					multiply = $SHARE$
				}
			}
		}
	}
}

	}

# Hobnob intent: the modifier does not stack, so the conclusion only lists a player's first grant per market per activity
gdp_schmooze_reset_conclusion_markets_effect = {
	$CHAR$ = {
		if = {
			limit = {
				is_ai = no
				NAND = {
					has_variable = gdp_schmooze_activity
					var:gdp_schmooze_activity = scope:activity
				}
			}
			set_variable = {
				name = gdp_schmooze_activity
				value = scope:activity
				years = 1
			}
			if = {
				limit = { has_variable_list = gdp_schmooze_conclusion_markets }
				clear_variable_list = gdp_schmooze_conclusion_markets
			}
		}
	}
}
