	type = character_event
	window = fullscreen_event

		triggered_desc = {
			trigger = { 
				is_guelph_leader_trigger = no

		first_valid = {
			triggered_desc = {
				trigger = { is_guelph_leader_trigger = yes }

			}
			triggered_desc = {
				trigger = { is_ghibelline_leader_trigger = yes }
				desc = gdp_guelph_ghibelline.0500.desc.ghibelline_leader

			triggered_desc = {
				trigger = { is_guelph_trigger = yes }
				desc = gdp_guelph_ghibelline.0500.desc.guelph_member
			}
			triggered_desc = {
				trigger = { is_ghibelline_trigger = yes }
				desc = gdp_guelph_ghibelline.0500.desc.ghibelline_member

	override_background = { reference = gdp_fullscreen_gg_start }
	cooldown = { days = 1 }

	widgets = {
		widget = {
 			gui = "event_window_widget_situation_info_guelph_ghibelline"
 			container = "dynamic_content_widget"
 			controller = situation_info
 			setup_scope = { situation:guelph_ghibelline = { save_scope_as = situation } }
		}
	}

	trigger = {
		is_ai = no
		NOT = { has_character_flag = had_gg_intro_event }
		OR = {
			has_variable = ghibelline_leader
			has_variable = guelph_leader
			any_held_title = {
				OR = {
					this = title:k_venice
					title_province = { geographical_region = gdp_guelph_ghibelline_northern_italy_region }
				}
			}
			top_overlord ?= {
				any_held_title = {
					OR = {
						this = title:k_venice
						title_province = { geographical_region = gdp_guelph_ghibelline_northern_italy_region }
					}
				}
			}
		}
	}

	immediate = {
		add_character_flag = had_gg_intro_event
		situation:guelph_ghibelline ?= {
			#Popero
			random_situation_participant = {
				limit = { 
					any_held_title = { this = title:k_papal_state }
				}
				save_scope_as = pope_scope
			}
			#Guelph Leader
			random_situation_participant = {
				limit = { has_variable = guelph_leader }
				save_scope_as = guelph_leader
			}
			#Ghibelline Leader
			random_situation_participant = {
				limit = { has_variable = ghibelline_leader }
				save_scope_as = ghibelline_leader
			}
		}
	}
	#:ohok:
	option = {
		name = gdp_guelph_ghibelline.0500.a
	}
}

#You have recruited a bunch of people to your side
gdp_guelph_ghibelline.1000 = {
	type = letter_event
	opening = gdp_guelph_ghibelline.1000.opening
	desc = gdp_guelph_ghibelline.1000.desc
	sender = {
		character = scope:gg_leader
		animation = admiration
	}

	immediate = {
		if = {
			limit = { is_guelph_trigger = yes }
			situation:guelph_ghibelline ?= { 
				random_situation_participant = {
					limit = { is_guelph_leader_trigger = yes }
					save_scope_as = gg_leader
				}
			}
		}
		else = {
			situation:guelph_ghibelline ?= { 
				random_situation_participant = {
					limit = { is_ghibelline_leader_trigger = yes }
					save_scope_as = gg_leader
				}
			}
		}
		#Randomize reward somewhat
		random_list = {
			10 = {
				dynasty = { 
					add_dynasty_prestige = {
						value = minor_dynasty_prestige_gain
						multiply = {
							add = {
								value = root.var:gg_involvement
								divide = 25
							}
						}
					}
				}
			}
			10 = {
				add_gold = {
					value = medium_gold_value
					multiply = {
						add = {
							value = var:gg_involvement
							divide = 25
						}
					}
				}
			}
			10 = {
				add_prestige = {
					value = medium_prestige_gain
					multiply = {
						add = {
							value = var:gg_involvement
							divide = 25
						}
					}
				}
			}
			10 = {
				trigger = { gg_has_medium_involvement_trigger = yes }
				if = {
					limit = { is_guelph_trigger = yes }
					add_character_modifier = {
						modifier = guelph_efforts_recognized_modifier
						years = 10
					}
				}
				else = {
					add_character_modifier = {
						modifier = ghibelline_efforts_recognized_modifier
						years = 10
					}
				}
			}
		}
	}
	#Yes, I did good didn't I!
	option = {
		name = gdp_guelph_ghibelline.1000.a
		scope:gg_leader = {
			add_hook = {
				target = root
				type = loyalty_hook
				years = 5
			}
		}
		if = {
			limit = { is_ai = yes }
			add_opinion = {
				target = scope:gg_leader
				modifier = feeling_appreciated_opinion
				opinion = 30
			}
		}
		if = {
			limit = {
				scope:gg_leader = { is_ai = yes }
			}
			scope:gg_leader = {
				add_opinion = {
					target = root
					modifier = appreciation_opinion
					opinion = 30
				}
			}
		}
		#And reset the counter
		remove_variable = recruited_guelphs_num
		remove_variable = recruited_ghibelline_num
	}
}

# Compromise!
gdp_guelph_ghibelline.8500 = {
	type = character_event
	window = fullscreen_event
	queue_icon = "gfx/interface/icons/situation_types/guelph_ghibelline.dds"
	title = gdp_guelph_ghibelline.8500.t
	desc = gdp_guelph_ghibelline.8500.desc
	theme = guelph_ghibelline
	override_background = { reference = gdp_fullscreen_gg_compromise }

	immediate = {
		if = {
			limit = {
				scope:actor = { has_variable = guelph_leader }
			}
			scope:actor = { save_scope_as = guelph_leader }
			scope:recipient = { save_scope_as = ghibelline_leader }
		}
		else = {
			scope:actor = { save_scope_as = ghibelline_leader }
			scope:recipient = { save_scope_as = guelph_leader }
		}
		if = {
			limit = { scope:actor = this }
			if = {
				limit = { has_variable = guelph_leader }
				gg_situation_compromise_actual_effect = { GUELPH_LEADER = scope:actor GHIBELLINE_LEADER = scope:recipient }
			}
			else = {
				gg_situation_compromise_actual_effect = { GUELPH_LEADER = scope:recipient GHIBELLINE_LEADER = scope:actor }
			}
			custom_description_no_bullet = {
				text = gg_end_situation_tt
				situation:guelph_ghibelline = { end_situation = yes }
			}
		}
		else = {
			show_as_tooltip = {
				if = {
					limit = { has_variable = guelph_leader }
					gg_situation_compromise_actual_effect = { GUELPH_LEADER = scope:actor GHIBELLINE_LEADER = scope:recipient }
				}
				else = {
					gg_situation_compromise_actual_effect = { GUELPH_LEADER = scope:recipient GHIBELLINE_LEADER = scope:actor }
				}
			}
			custom_description_no_bullet = { text = gg_end_situation_tt }
		}
	}
	#:ohok:
	option = {
		name = gdp_guelph_ghibelline.8500.a
		custom_description_no_bullet = {
			text = gg_compromise_leader_involvement_effect
			dynasty ?= { add_dynasty_prestige = root.gg_compromise_leader_involvement_renown }
		}
	}
	after = {
		if = {
			limit = { scope:actor = this }
			#Actually end the situation
			trigger_event = {
				id = gdp_guelph_ghibelline.9001
				days = 1
			}
		}
	}
}
