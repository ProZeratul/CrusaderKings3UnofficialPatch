# benevolent_charter_authority_decision

scripted_effect merchant_charter_benevolent_reward_effect = {
	if = {
		limit = {
			has_hook_of_type = {
				target = scope:patron
				type = indebted_hook
			}
		}
		hidden_effect = {
			remove_hook = {
				target = scope:patron
				type = indebted_hook
			}
		}
		add_hook = {
			target = scope:patron
			type = strong_indebted_hook
		}
	}
	else_if = {
		limit = {
			can_add_hook = {
				target = scope:patron
				type = indebted_hook
			}
		}
		add_hook = {
			target = scope:patron
			type = indebted_hook
		}
	}
	else = { add_gold = medium_gold_value }
	if = {
		limit = { is_ai = yes }
		add_opinion = {
			target = scope:patron
			modifier = grateful_opinion
			opinion = 15
		}
	}
}

gdp_decision.3000 = {
	type = letter_event
	opening = gdp_decision.3000.opening
	desc = gdp_decision.3000.desc
	sender = scope:new_patron

	trigger = {
		exists = domicile
		government_has_flag = government_is_merchant
		trigger_if = {
			limit = { var:enticed_merchant ?= var:enticed_merchant } # error suppression
		}
	}

	immediate = {
		save_scope_as = new_client
		add_to_variable_list = {
			name = entice_merchant_cooldown
			target = scope:new_patron
			years = 10
		}
		if = {
			limit = { is_ai = no }
			add_character_flag = {
				flag = entice_merchant_invited
				months = 4
			}
		}
		# Select arrival location
		scope:new_patron ?= {
			capital_province ?= {
				save_scope_as = new_domicile_location
				barony.trade_market ?= { save_scope_as = new_market }
			}
		}
	}

	option = {
		name = gdp_decision.3000.a
		# Move warnings
		if = {
			limit = {
				exists = scope:new_market
				scope:new_market = {
					any_character_in_market = {
						NOR = {
							this = root
							house ?= root.house
						}
						trade_competitiveness > root.trade_competitiveness
					}
				}
			}
			custom_tooltip = move_company_interaction.low_competitiveness
		}
		gdp_move_company_invalidate_trade_effect = yes
		if = {
			limit = {
				exists = scope:new_patron
				NOT = { suzerain ?= scope:new_patron }
			}
			custom_tooltip = move_company_interaction.patron_alert
		}
		if = {
			limit = {
				exists = scope:new_patron
				trigger_if = {
					limit = { has_realm_law = company_size_huge }
					scope:new_patron.highest_held_title_tier < tier_kingdom
				}
				trigger_else_if = {
					limit = { has_realm_law = company_size_large }
					scope:new_patron.highest_held_title_tier < tier_duchy
				}
				trigger_else = { always = no }
			}
			custom_tooltip = move_company_interaction.size_alert
		}
		add_internal_flag = special
		# Add cooldown
		add_character_flag = {
			flag = entice_merchant_cooldown
			years = 10
		}
		# End Patron relationship and send message if relevant
		suzerain = {
			save_scope_as = old_patron
			if = {
				limit = {
					is_ai = no
					NOT = { this = scope:new_patron }
				}
				send_interface_toast = {
					type = event_merchant_bad
					title = merchant_enticed_title
					left_icon = scope:new_client
					right_icon = scope:new_client.primary_title
					scope:new_client = { end_tributary = yes }
				}
			}
		}
		# Inform new Patron and grant enticements
		if = {
			limit = {
				exists = scope:new_client
				exists = scope:new_domicile_location
			}
			scope:new_client = {
				send_interface_message = {
					type = msg_merchant_enticed
					title = merchant_enticed_title 
					left_icon = scope:new_patron
					right_icon = scope:new_domicile_location.county
					scope:new_patron ?= {
						send_interface_message = {
							type = msg_merchant_enticed
							title = merchant_enticed_title 
							left_icon = scope:new_client
							right_icon = scope:new_client.primary_title
							scope:new_client = {
								add_short_term_gold = scope:new_patron.minor_gold_value
								add_prestige = medium_prestige_value
								domicile = { move_domicile = scope:new_domicile_location }
								# Start new Patron relationship
								hidden_effect = {
									start_tributary_interaction_effect = {
										TRIBUTARY = scope:new_client
										SUZERAIN = scope:new_patron
									}
								}
								scope:new_market = {
									add_competitiveness_modifier_for_character = {
										character = scope:new_client
										modifier = gdp_enticed_merchant_market_modifier
										years = 5
									}
								}
							}
							if = {
								limit = {
									scope:old_patron = {
										is_ai = yes
										is_alive = yes
									}
									NOT = { this = scope:old_patron }
								}
								reverse_add_opinion = {
									target = scope:old_patron
									modifier = annoyed_opinion
									opinion = -15
								}
							}
						}
					}
				}
			}
			if = {
				limit = {
					scope:old_patron = {
						is_ai = yes
						is_alive = yes
					}
					NOT = { scope:new_patron = scope:old_patron }
					NOT = { this = scope:old_patron }
