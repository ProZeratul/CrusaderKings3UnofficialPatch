## What if the character has a trait that would make it easier/harder for them to convince the target?
scripted_effect influence_house_ambition_trait_bonus_effect = {
	## Character hears voices and rambles insanely
	if = {
		limit = { 
			OR = { 
				has_trait = possessed_1
				has_trait = lunatic_1
			} 
		}
		scope:scheme = {
			add_scheme_progress = {
				value = scheme_progress_loss
				multiply = -30
			}
			add_scheme_modifier = { type = gdp_insane_arguments_modifier }
		}
	}
	## Character is depressed
	if = {
		limit = { has_trait = depressed_1 }
		scope:scheme = {
			add_scheme_modifier = { type = gdp_fatalistic_arguments_modifier }
		}
	}
	## Character is easy to look at
	if = {
		limit = { has_trait = beauty_good_1 }
		scope:scheme = {
			add_scheme_modifier = { type = gdp_beautiful_arguments_modifier }
		}
	}
	## Character is hard to look at
	if = {
		limit = { has_trait = beauty_bad_1 }
		scope:scheme = {
			add_scheme_modifier = { type = gdp_ugly_arguments_modifier }
		}
	}
	## Character is Smart!
	if = {
		limit = { has_trait = intellect_good_1 }
		scope:scheme = {
			add_scheme_progress = {
				value = scheme_progress_loss
				multiply = -30
			}
			add_scheme_modifier = { type = gdp_intellectual_debate_modifier }
		}

	}
	## Character is kind of dumb
	if = {
		limit = { has_trait = intellect_bad_1 }
		scope:scheme = {
			add_scheme_modifier = { type = gdp_advise_of_a_fool_modifier }
		}
	}
	## Character has a way with words
	if = {
		limit = { 
			OR = { 
				has_trait = lifestyle_mystic
				has_trait = theologian
				has_trait = erudite
			} 
		}
		scope:scheme = {
			add_scheme_modifier = { type = gdp_a_way_with_words_modifier }
		}
	}
}
## What do they want in return?
scripted_effect influence_house_ambition_prize_effect = {
	this = {
		save_scope_as = payed_agent_01
	}
	if = {
		limit = {
			has_trait = greedy
		}
		root = {
			pay_short_term_gold = {
				target = scope:payed_agent_01
				gold = medium_gold_value
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_relation_friend = root
				has_relation_lover = root
				is_close_family_of = root
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = lustful
				has_trait = seducer
			}
			is_attracted_to_gender_of = root
			might_cheat_on_every_partner_trigger = yes
		}
		progress_towards_lover_effect = {
			CHARACTER = root
			REASON = seduced_at_competition
			OPINION = default_lover_opinion
		}
		had_sex_with_effect = {
			CHARACTER = root
			PREGNANCY_CHANCE = pregnancy_chance
		}
	}
	else = {
		add_hook = {
			target = root
			type = favor_hook
		}
	}
}

scripted_effect influence_house_ambition_intimidation_response_effect = {
	if = {
		limit = {
			has_trait = vengeful
		}
		set_relation_rival_effect = {
		    TARGET = root
		    REASON = rival_intimidation
		}
	}
	else_if = {
		limit = {
			has_trait = brave
		}
		scope:scheme = { add_scheme_modifier = { type = gdp_hardened_position_modifier } }
		progress_towards_rival_effect = {
			REASON = rival_clothes_stolen
			CHARACTER = root
			OPINION = -15
		}
	}
	else_if = {
		limit = {
			has_trait = craven
		}
		scope:scheme = { add_scheme_modifier = { type = gdp_intimidated_modifier } }
		add_opinion = {
			modifier = threatened_opinion
			target = root
			opinion = -5
		}
	}
	else = {
		add_opinion = {
			modifier = threatened_opinion
			target = root
			opinion = -10
		}	
	}
}
